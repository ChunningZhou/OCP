"""Which languages a cipher's implementation can be emitted in, built in, and KAT-checked in."""

import ast
import inspect

import implementations.implementations as imp

# What a toolchain leaves behind beside the emitted source.
BUILD_PRODUCTS = ("out", "dll")

# Per language: artifact extension, toolchain probe (None when nothing has to be installed), and
# KAT entry point. Whether that entry point checks anything is derived, not declared.
_LANGUAGES = {
    "python": {"extension": "py", "probe": None, "tester": "test_implementation_python"},
    "c": {"extension": "c", "probe": "is_c_compiler_available", "tester": "test_implementation_c"},
    "verilog": {"extension": "sv", "probe": "is_verilog_compiler_available",
                "tester": "test_implementation_verilog"},
    "rust": {"extension": "rs", "probe": "is_rust_compiler_available",
             "tester": "test_implementation_rust"},
}


def _tester_returns(tester_name):
    """The values the tester can return, read from its source; a computed one reads as ``"expr"``.

    Only returning ``None`` means it checks nothing, which is how ``automated_kat`` is decided.
    """
    tree = ast.parse(inspect.getsource(getattr(imp, tester_name)).lstrip())
    return {node.value.value if isinstance(node.value, ast.Constant) else "expr"
            for node in ast.walk(tree) if isinstance(node, ast.Return)}


def _tester_runs_artifact(tester_name):
    """Whether the tester actually executes what was emitted, read from its source.

    A toolchain on PATH says a thing can be built, not that it runs here, and these are separate
    facts: ``test_implementation_c`` hands the compiled binary to ``subprocess.run``,
    ``test_implementation_python`` evaluates the generated module, and
    ``test_implementation_verilog`` prints the command it would use and runs nothing.
    """
    tree = ast.parse(inspect.getsource(getattr(imp, tester_name)).lstrip())
    for node in ast.walk(tree):
        if not isinstance(node, ast.Call):
            continue
        name = getattr(node.func, "attr", None) or getattr(node.func, "id", "")
        if name in ("run", "check_output", "Popen") or name.startswith("evaluate_"):
            return True
    return False

def languages():
    """``{language: {...}}`` for every declared language, at repository level.

    Five questions kept apart on purpose: ``emits`` (the dispatch writes it),
    ``compiler_available`` (the tool is on PATH, ``None`` when nothing can be emitted),
    ``runs_artifact`` (the tester executes what was emitted rather than describing how to),
    ``automated_kat`` (the tester compares against a vector -- **False for verilog**, whose tester
    prints a command and returns ``None``), and ``kat_returns`` (``True`` matched, ``False`` did
    not, ``None`` skipped or unchecked). Having a tester is not having been verified.

    For one cipher use ``cipher_languages``: an operator can support fewer languages than this.
    """
    report = {}
    for name, entry in _LANGUAGES.items():
        emits = name in imp.IMPLEMENTED_LANGUAGES
        if not emits:
            available, tool = None, None
        elif entry["probe"] is None:
            available, tool = True, None      # python needs nothing installed
        else:
            available, tool = getattr(imp, entry["probe"])()
        returns = _tester_returns(entry["tester"])
        report[name] = {
            "extension": entry["extension"],
            "emits": emits,
            "compiler_available": available,
            "tool": tool,
            # None, like compiler_available, when there is no artifact for a tester to check.
            "automated_kat": (returns != {None}) if emits else None,
            # Separate from compiler_available: built is not run. False for verilog, whose tester
            # prints the command and executes nothing.
            "runs_artifact": _tester_runs_artifact(entry["tester"]),
            "kat_returns": sorted(returns, key=repr),
            "tester": entry["tester"],
        }
    return report

def cipher_languages(cipher):
    """The languages every operator of ``cipher`` declares, intersected.

    Walks what emission walks, not the model scope. 7 of 32 ciphers lose verilog this way.
    """
    supported = None
    for function in cipher.functions.values():
        for round_index in range(1, function.nbr_rounds + 1):
            for layer in range(function.nbr_layers + 1):
                for cons in function.constraints[round_index][layer]:
                    declared = set(getattr(type(cons), "SUPPORTED_IMPLEMENTATIONS", ()) or ())
                    if declared:
                        supported = declared if supported is None else supported & declared
    return sorted(supported or ())

def feasible_languages(cipher=None, require_compiler=True, require_kat=False):
    """The languages an implementation can be emitted in; without ``cipher``, repository-wide.

    ``require_compiler`` also demands the build tool, ``require_kat`` a tester that compares.
    """
    allowed = None if cipher is None else set(cipher_languages(cipher))
    return [name for name, entry in languages().items()
            if entry["emits"]
            and (allowed is None or name in allowed)
            and (entry["compiler_available"] or not require_compiler)
            and (entry["automated_kat"] or not require_kat)]

def source_extensions():
    """``{language: extension}`` for the source file each language emits."""
    return {name: entry["extension"] for name, entry in _LANGUAGES.items()}

def artifact_extensions():
    """Sources and build products together; ``source_extensions()`` alone to pick a source file."""
    return tuple(sorted(set(source_extensions().values()) | set(BUILD_PRODUCTS)))




# One code per thing that can stand in the way, so a consumer branches on a value rather than on
# English. `host` marks the ones that describe this machine and not the repository.
BLOCKERS = {
    "unknown_language": {"host": False, "detail": "not a language this layer knows"},
    "not_implemented": {"host": False, "detail": "declared but generate_implementation raises"},
    "operators_lack_language": {"host": False,
                                "detail": "an operator of this cipher does not declare it"},
    "no_toolchain": {"host": True, "detail": "no build tool on PATH here"},
    "no_automated_kat": {"host": False, "detail": "the tester compares nothing and returns None"},
}

def language_report(language, cipher=None):
    """``{language, available, blockers: [...], ...}`` -- **every** reason, not just the first.

    ``explain_language`` used to stop at the first problem, so AES and verilog reported only that
    the operators do not declare it; fixing that would have revealed a missing toolchain, and then
    a tester that checks nothing. Three separate facts behind one message.

    Each blocker carries a ``code`` from ``BLOCKERS`` to branch on and a ``host`` flag: a missing
    toolchain says something about this machine, the other four about the repository.
    """
    entry = languages().get(language)
    if entry is None:
        return {"language": language, "available": False, "known": False,
                "blockers": [dict(code="unknown_language", host=False,
                                  detail=f"expected one of {sorted(_LANGUAGES)}")]}
    blockers = []
    if not entry["emits"]:
        blockers.append(dict(code="not_implemented", host=False,
                             detail="generate_implementation raises for it"))
    if cipher is not None and language not in cipher_languages(cipher):
        blockers.append(dict(code="operators_lack_language", host=False,
                             detail=f"{cipher.name}'s operators declare only "
                                    f"{cipher_languages(cipher)}"))
    if entry["emits"] and not entry["compiler_available"]:
        blockers.append(dict(code="no_toolchain", host=True,
                             detail="no build tool on PATH; the artifact would be written and "
                                   "never checked against a test vector"))
    if entry["emits"] and not entry["automated_kat"]:
        blockers.append(dict(code="no_automated_kat", host=False,
                             detail=f"{entry['tester']} returns None on every path"))
    return {"language": language, "available": not blockers, "known": True,
            "blockers": blockers, "emits": entry["emits"],
            "compiler_available": entry["compiler_available"],
            "automated_kat": entry["automated_kat"]}

def explain_language(language, cipher=None):
    """Why ``language`` cannot be used; ``""`` when it can. A rendering of ``language_report``.

    Lists every blocker rather than the first, so solving one does not reveal another.
    """
    report = language_report(language, cipher)
    if report["available"]:
        return ""
    listed = "; ".join(f"{b['code']} ({b['detail']})" for b in report["blockers"])
    return f"[ERROR] Language {language} cannot be used: {listed}."

def read_write_asymmetry():
    """The one trap the signatures hide, which no other function here reports."""
    return ("generate_implementation(prim, filename, ...) writes where the caller says; "
            "test_implementation_*(cipher, cipher_name, ...) reads "
            "_generated_path(cipher_name, ext). Pass a filename whose stem is cipher_name, or "
            "the tester looks in the wrong place.")
