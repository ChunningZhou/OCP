"""What each operator declares it can model and emit, cross-checked against its source."""

import ast
import importlib
import inspect
import os

from operators.operators import NoneOperator, Operator

# Operators that emit nothing, so nothing can be blocked by them. The CLASSES, not their names:
# a name would also exempt same-named and derived classes that do emit.
NO_MODEL_OPERATORS = frozenset({NoneOperator})

# The capability surface is SUPPORTED_MODEL_VERSIONS plus SUPPORTED_IMPLEMENTATIONS.
_OPERATORS_PACKAGE = "operators"

def _modules(package_name, skip, include_in_development):
    """Import and return the package's modules in filename order, skipping unfinished ones."""
    modules = []
    package = importlib.import_module(package_name)
    for filename in sorted(os.listdir(os.path.dirname(inspect.getfile(package)))):
        if not filename.endswith(".py") or filename == "__init__.py" or filename in skip:
            continue
        module_name = f"{package_name}.{filename[:-3]}"
        try:
            module = importlib.import_module(module_name)
        except Exception as exc:
            raise RuntimeError(f"[ERROR] Cannot enumerate {package_name}: importing {module_name} failed: {exc}") from exc
        if getattr(module, "IN_DEVELOPMENT", False) and not include_in_development:
            continue
        modules.append(module)
    return modules

def operator_classes(declaring_only=False, include_in_development=False):
    """``{class name: class}`` for every concrete ``Operator`` subclass under ``operators/``.

    ``declaring_only=True`` keeps the classes with a non-empty declaration, and ``{"milp": ()}``
    counts -- empty version tuples are a statement. Keyed by ``__name__``; two classes sharing it
    raise rather than overwrite.
    """
    found, origin = {}, {}
    for module in _modules(_OPERATORS_PACKAGE, (), include_in_development):
        for _, operator_class in inspect.getmembers(module, inspect.isclass):
            if not (operator_class.__module__ == module.__name__ and issubclass(operator_class, Operator)
                    and not getattr(operator_class, "__abstractmethods__", None)):
                continue
            name = operator_class.__name__
            if name in origin:
                raise RuntimeError(
                    f"[ERROR] Two operator classes are named {name}: {origin[name]} and "
                    f"{operator_class.__module__}. Operator capabilities are keyed by class name, "
                    f"so rename one of them."
                )
            origin[name] = operator_class.__module__
            if not declaring_only or getattr(operator_class, "SUPPORTED_MODEL_VERSIONS", None):
                found[name] = operator_class
    return found

def describe_operators(include_in_development=False):
    """``{operator: {module, model_versions, implementations}}`` from the declarations themselves.

    Only classes declaring ``SUPPORTED_MODEL_VERSIONS`` in their own ``__dict__``; one that
    inherits it (every named S-box inheriting ``Sbox``) is represented by the declaring base.
    ``model_versions`` is the class's own declaration, ``implementations`` the effective value
    after inheritance, and ``implementations_declared_here`` says whether it overrides.
    """
    result = {}
    for name, operator_class in operator_classes(include_in_development=include_in_development).items():
        if "SUPPORTED_MODEL_VERSIONS" not in operator_class.__dict__:
            continue
        declaration = operator_class.__dict__["SUPPORTED_MODEL_VERSIONS"]
        result[name] = {
            "module": operator_class.__module__.split(".")[-1] + ".py",
            "model_versions": {model_type: list(versions) for model_type, versions in declaration.items()},
            "implementations": list(getattr(operator_class, "SUPPORTED_IMPLEMENTATIONS", ())),
            "implementations_declared_here": "SUPPORTED_IMPLEMENTATIONS" in operator_class.__dict__,
        }
    return dict(sorted(result.items()))

def delegates(operator_class):
    """Whether the class declares it passes its ``model_version`` down to inner operators."""
    return bool(getattr(operator_class, "MODEL_VERSIONS_DELEGATED", False))

def models_nothing(operator_class):
    """Whether ``operator_class`` declares every model_type with no supported versions."""
    declaration = operator_class.__dict__.get("SUPPORTED_MODEL_VERSIONS")
    return bool(declaration) and not any(declaration.values())

def _model_version_suffixes(node):
    suffixes = set()
    for sub in ast.walk(node):
        if isinstance(sub, ast.BinOp) and isinstance(sub.op, ast.Add):
            left, right = sub.left, sub.right
            if isinstance(right, ast.Constant) and isinstance(right.value, str) and right.value.startswith("_"):
                is_class_name = (
                    (isinstance(left, ast.Attribute) and left.attr == "__name__")
                    or (isinstance(left, ast.Name) and left.id == "class_name")
                )
                if is_class_name:
                    suffixes.add(right.value[1:])
    return suffixes


def _called_module_funcs(node, module_funcs):
    return {
        sub.func.id
        for sub in ast.walk(node)
        if isinstance(sub, ast.Call) and isinstance(sub.func, ast.Name) and sub.func.id in module_funcs
    }


def _generate_model_defs():
    """``{class name: (generate_model ast node, module functions)}`` parsed from ``operators/``."""
    found = {}
    package = importlib.import_module(_OPERATORS_PACKAGE)
    operators_dir = os.path.dirname(inspect.getfile(package))
    for filename in sorted(os.listdir(operators_dir)):
        if not filename.endswith(".py") or filename == "__init__.py":
            continue
        with open(os.path.join(operators_dir, filename), encoding="utf-8") as handle:
            tree = ast.parse(handle.read())
        module_funcs = {n.name: n for n in tree.body if isinstance(n, ast.FunctionDef)}
        for cls in [n for n in tree.body if isinstance(n, ast.ClassDef)]:
            gm = next((m for m in cls.body if isinstance(m, ast.FunctionDef) and m.name == "generate_model"), None)
            if gm is not None:
                found[cls.name] = (gm, module_funcs)
    return found

def extract_from_source():
    """``{operator: set(model_version_suffixes)}`` statically parsed out of ``generate_model``."""
    result = {}
    for name, (gm, module_funcs) in _generate_model_defs().items():
        versions = _model_version_suffixes(gm)
        for func_name in _called_module_funcs(gm, module_funcs):
            versions |= _model_version_suffixes(module_funcs[func_name])
        if versions:
            result[name] = versions
    return result

def source_models_nothing():
    """The operators whose ``generate_model`` cannot return anything, parsed statically.

    A heuristic, not an analysis: the converse is weaker, so the proof is a test that calls them.
    """
    return {name for name, (gm, _) in _generate_model_defs().items()
            if not any(isinstance(n, ast.Return) and n.value is not None for n in ast.walk(gm))}




def _implementation_languages(fn):
    """The languages an ``implementation_type == <literal>`` comparison names inside ``fn``.

    Comparisons only: every string constant would include the ``implementation_type='python'``
    default argument, making an operator that handles nothing look like it supports python.
    """
    languages = set()
    for node in ast.walk(fn):
        if isinstance(node, ast.Compare) and isinstance(node.left, ast.Name) and node.left.id == "implementation_type":
            for comparator in node.comparators:
                for sub in ast.walk(comparator):
                    if isinstance(sub, ast.Constant) and isinstance(sub.value, str):
                        languages.add(sub.value)
    return languages

def extract_implementations_from_source():
    """``{operator: set(languages)}`` that ``generate_implementation`` branches on, statically."""
    result = {}
    package = importlib.import_module(_OPERATORS_PACKAGE)
    operators_dir = os.path.dirname(inspect.getfile(package))
    for filename in sorted(os.listdir(operators_dir)):
        if not filename.endswith(".py") or filename == "__init__.py":
            continue
        with open(os.path.join(operators_dir, filename), encoding="utf-8") as handle:
            tree = ast.parse(handle.read())
        for cls in [n for n in tree.body if isinstance(n, ast.ClassDef)]:
            gi = next((m for m in cls.body if isinstance(m, ast.FunctionDef)
                       and m.name == "generate_implementation"), None)
            if gi is not None:
                result[cls.name] = _implementation_languages(gi)
    return result

def consistency_report(include_in_development=False):
    """Diff the declarations against the source; ``{operator: {...}}``, empty when they agree.

    Per operator, any of: ``missing`` (the code handles it but the declaration omits it -- real
    drift), ``extra`` (declared but not seen statically, usually benign), ``declares_no_model`` /
    ``source_has_no_model``, ``implementations_missing`` / ``implementations_extra``. An empty
    report does **not** prove the declarations; ``verification_coverage()`` says what it reaches.
    """
    described = describe_operators(include_in_development=include_in_development)
    declared = {
        name: set().union(*(set(versions) for versions in info["model_versions"].values()))
        if info["model_versions"] else set()
        for name, info in described.items()
    }
    extracted = extract_from_source()
    implemented = extract_implementations_from_source()
    no_model_in_source = source_models_nothing()
    classes = operator_classes(include_in_development=include_in_development)

    report = {}
    # Over EVERY concrete class, not just the declaring ones: an operator that returns nothing is
    # exactly the one that declares nothing, which a loop over the declarations would skip.
    for name, operator_class in classes.items():
        entry = {}
        if name in described:
            missing = extracted.get(name, set()) - declared[name]
            extra = set() if delegates(operator_class) else declared[name] - extracted.get(name, set())
            if missing:
                entry["missing"] = sorted(missing)
            if extra:
                entry["extra"] = sorted(extra)
        declares_none = models_nothing(operator_class)
        if declares_none and name not in no_model_in_source:
            entry["declares_no_model"] = "every model_type declared empty, but generate_model returns a value"
        if not declares_none and name in no_model_in_source:
            entry["source_has_no_model"] = "every generate_model branch raises; declare empty version tuples"

        # An operator that emits nothing names no language: absence of evidence, not a mismatch.
        source_languages = implemented.get(name)
        if source_languages and operator_class not in NO_MODEL_OPERATORS:
            declared_languages = set(getattr(operator_class, "SUPPORTED_IMPLEMENTATIONS", ()) or ())
            if source_languages - declared_languages:
                entry["implementations_missing"] = sorted(source_languages - declared_languages)
            if declared_languages - source_languages:
                entry["implementations_extra"] = sorted(declared_languages - source_languages)
        if entry:
            report[name] = entry
    return report

def verification_coverage():
    """What ``consistency_report()`` actually proves, and what it leaves unverified."""
    described = describe_operators()
    extracted = extract_from_source()
    implemented = extract_implementations_from_source()
    classes = operator_classes()

    declared_versions = sum(
        len(set().union(*(set(v) for v in info["model_versions"].values())) if info["model_versions"] else set())
        for info in described.values()
    )
    confirmed_versions = sum(
        len((set().union(*(set(v) for v in info["model_versions"].values())) if info["model_versions"] else set())
            & extracted.get(name, set()))
        for name, info in described.items()
    )
    delegated = sorted(name for name, cls in classes.items() if delegates(cls))
    languages_unconfirmable = sorted(
        name for name in described
        if not implemented.get(name) and classes[name] not in NO_MODEL_OPERATORS
    )
    return {
        "versions_declared": declared_versions,
        "versions_confirmed_in_source": confirmed_versions,
        "versions_delegated": delegated,
        "per_model_type_split": (
            "NOT VERIFIED. The diff is over the union of the model types, and it cannot be "
            "otherwise: most operators put their version branches under a combined "
            "`if model_type in ['sat', 'milp']`, so the source does not say which backend a "
            "version belongs to. A version moved between backends, or dropped from one, passes."
        ),
        "languages_unconfirmable": languages_unconfirmable,
        "generate_model_bodies": (
            "NOT ANALYSED. Versions are recognised only as `self.__class__.__name__ + \"_X\"` "
            "string concatenation, and helper calls only one level deep into module-level "
            "functions of the same file. A dynamically built version, or one reached through a "
            "method helper or a second level of delegation, is invisible here."
        ),
    }
