"""The layer methods a cipher is built from: how each is called and what each argument must be."""

import ast
import importlib
import inspect
import os
import re

import primitives.primitives as prim

# Read out of primitives/primitives.py, by two different means with different strength. The method
# set, the signatures and the operator parameters are DERIVED -- they cannot disagree with the
# code. The per-argument descriptions are PROSE lifted from each method's ``Args:`` block, and
# `accepted_values` / `validates` are literals scraped out of comparisons and raises: those can be
# incomplete or stale, so the report marks what it could not establish rather than implying it.
_LAYERED_CLASS = "Layered_Function"
_PRIMITIVES_PACKAGE = "primitives"

_ARG_LINE = re.compile(r"^(\w+)\s*(\([^)]*\))?\s*:\s*(.*)$")


def _layered_function_ast():
    """The ``Layered_Function`` class node, and the source lines it was parsed from."""
    source = inspect.getsource(prim)
    tree = ast.parse(source)
    node = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == _LAYERED_CLASS)
    return node, source.split("\n")


def _purpose(method, lines):
    """Its docstring's first line, else the comment above it."""
    doc = ast.get_docstring(method)
    if doc:
        return doc.split("\n")[0].strip()
    comment = []
    for index in range(method.lineno - 2, -1, -1):
        stripped = lines[index].strip()
        if not stripped.startswith("#"):
            break
        comment.insert(0, stripped.lstrip("#").strip())
    return " ".join(comment)


def _argument_docs(method):
    """``{parameter: prose}`` from the method's ``Args:`` block; empty when it has none.

    Prose, not a derived fact: a parameter missing here is undocumented in the code, which
    ``layer_methods`` reports rather than papering over.
    """
    doc = ast.get_docstring(method) or ""
    if "Args:" not in doc:
        return {}
    body = doc.split("Args:", 1)[1]
    for stop in ("\nReturns:", "\nExample:", "\nRaises:"):
        body = body.split(stop, 1)[0]
    docs, current = {}, None
    for line in body.split("\n"):
        stripped = line.strip()
        if not stripped:
            continue
        match = _ARG_LINE.match(stripped)
        if match and match.group(1) not in ("Example", "Returns"):
            current = match.group(1)
            docs[current] = match.group(3).strip()
        elif current:
            docs[current] += " " + stripped
    return docs


def _argument_shape(node):
    """How an argument reads in the source: its name, or ``[n]`` for an n-element list literal."""
    if isinstance(node, ast.Name):
        return node.id
    if isinstance(node, ast.List):
        return f"[{len(node.elts)} vars]"
    return "..."


def _operator_parameters(method):
    """``{parameter: called signature}`` for parameters the method calls as a constructor.

    Derived, not guessed from the name: ``sbox_operator`` and ``my_operator`` appear in
    ``Call.func`` position, so the report can say which argument is an operator class and how it
    will be called -- which a bare ``creates: []`` hid.
    """
    names = {argument.arg for argument in method.args.args}
    called = {}
    for node in ast.walk(method):
        if isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and node.func.id in names:
            positional = ", ".join(_argument_shape(a) for a in node.args)
            keywords = ", ".join(f"{k.arg}=..." for k in node.keywords if k.arg)
            called[node.func.id] = f"({', '.join(p for p in (positional, keywords) if p)})"
    return called


def _created_operators(method):
    """The operator classes the method instantiates by name; see ``_operator_parameters``."""
    created = set()
    for node in ast.walk(method):
        if isinstance(node, ast.Call):
            name = node.func.attr if isinstance(node.func, ast.Attribute) else getattr(node.func, "id", "")
            if name[:1].isupper() and not name.endswith("Error"):
                created.add(name)
    return sorted(created)


def _accepted_values(method):
    """``{parameter: [literals]}`` scraped from ``parameter == "literal"`` comparisons.

    Only as complete as that pattern: a parameter checked another way reports nothing, so an
    empty result is not evidence that anything goes. A test calls the ones reported.
    """
    accepted = {}
    for node in ast.walk(method):
        if isinstance(node, ast.Compare) and isinstance(node.left, ast.Name):
            for comparator in node.comparators:
                if isinstance(comparator, ast.Constant) and isinstance(comparator.value, str):
                    accepted.setdefault(node.left.id, set()).add(comparator.value)
    return {name: sorted(values) for name, values in accepted.items()}


def _validations(method):
    """The messages it raises. Message text, not a machine-readable condition."""
    messages = []
    for node in ast.walk(method):
        if not isinstance(node, ast.Raise) or node.exc is None:
            continue
        for sub in ast.walk(node.exc):
            if isinstance(sub, ast.Constant) and isinstance(sub.value, str) and len(sub.value) > 4:
                messages.append(sub.value.strip())
    return sorted(set(messages))


def _module_usage():
    """``{layer method: how many modules under primitives/ contain a call to it}``."""
    counts = {}
    package = importlib.import_module(_PRIMITIVES_PACKAGE)
    directory = os.path.dirname(inspect.getfile(package))
    for filename in sorted(os.listdir(directory)):
        if not filename.endswith(".py") or filename in ("__init__.py", "primitives.py"):
            continue
        with open(os.path.join(directory, filename), encoding="utf-8") as handle:
            tree = ast.parse(handle.read())
        called = {node.func.attr for node in ast.walk(tree)
                  if isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute)
                  and node.func.attr.endswith("Layer")}
        for name in called:
            counts[name] = counts.get(name, 0) + 1
    return counts

def layer_examples():
    """``{layer method: {module, line, call}}`` -- the first real call to each, from a cipher file.

    Lifted from ``primitives/*.py`` rather than written by hand, so an example cannot rot into
    something that no longer works: it *is* a line a shipped cipher runs. The ``Args:`` prose says
    what each argument means, which is not the same as showing the shape -- ``rot`` reads
    ``[['l', 16, 3, 3], ...]`` and no sentence conveys that as directly.

    A method no cipher calls has no entry; ``unused_layers()`` names those. The call is sliced by
    column, so an example inside an ``if`` gives the call alone rather than the statement.
    """
    examples = {}
    names = set(_layer_method_names())
    package = importlib.import_module(_PRIMITIVES_PACKAGE)
    directory = os.path.dirname(inspect.getfile(package))
    for filename in sorted(os.listdir(directory)):
        if not filename.endswith(".py") or filename in ("__init__.py", "primitives.py"):
            continue
        path = os.path.join(directory, filename)
        with open(path, encoding="utf-8") as handle:
            source = handle.read()
        lines = source.split("\n")
        for node in ast.walk(ast.parse(source)):
            if not (isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute)
                    and node.func.attr in names and node.func.attr not in examples):
                continue
            segment = "\n".join(lines[node.lineno - 1:node.end_lineno])
            start = node.col_offset
            end = node.end_col_offset if node.lineno == node.end_lineno else None
            call = segment[start:end] if end is not None else segment[start:]
            examples[node.func.attr] = {"module": filename, "line": node.lineno,
                                        "call": " ".join(call.split())}
    return examples




def _layer_method_names():
    """The ``*Layer`` method names on ``Layered_Function``."""
    node, _ = _layered_function_ast()
    return [m.name for m in node.body
            if isinstance(m, ast.FunctionDef) and m.name.endswith("Layer")]

def receiver_note():
    """The one thing every example hides: what the method is called on."""
    return ("A layer method is called on a Layered_Function -- cipher.functions['PERMUTATION'] "
            "or ['KEY_SCHEDULE'] / ['SUBKEYS'] -- not on the cipher. The examples show that "
            "receiver as the local name the cipher file gave it (S, KS, SK).")

def layer_methods():
    """``{name: {...}}`` for every ``*Layer`` method of ``Layered_Function``.

    Per entry: ``purpose``; ``parameters`` as ``{name: {required, default, description}}`` where
    the description is the ``Args:`` prose or ``""``; ``undocumented`` listing the parameters with
    no prose; ``operator_parameters`` (argument -> how it is called as a constructor);
    ``creates`` (classes named outright -- ask the C axis what they support);
    ``accepted_values`` and ``validates``, both scraped and possibly incomplete;
    ``primitive_modules_using``, a count of modules containing a call, not of calls or ciphers;
    and ``example``, a real call lifted from a cipher file (``None`` when nothing calls it).
    """
    node, lines = _layered_function_ast()
    usage = _module_usage()
    examples = layer_examples()
    methods = {}
    for method in node.body:
        if not (isinstance(method, ast.FunctionDef) and method.name.endswith("Layer")):
            continue
        signature = inspect.signature(getattr(prim.Layered_Function, method.name))
        described = _argument_docs(method)
        parameters = {
            name: {
                "required": value.default is inspect.Parameter.empty,
                "default": None if value.default is inspect.Parameter.empty else value.default,
                "description": described.get(name, ""),
            }
            for name, value in signature.parameters.items() if name != "self"
        }
        methods[method.name] = {
            "purpose": _purpose(method, lines),
            "parameters": parameters,
            "undocumented": sorted(n for n, p in parameters.items() if not p["description"]),
            "operator_parameters": _operator_parameters(method),
            "creates": _created_operators(method),
            "accepted_values": _accepted_values(method),
            "validates": _validations(method),
            "primitive_modules_using": usage.get(method.name, 0),
            "example": examples.get(method.name),
        }
    return methods

def describe_layer(name):
    """One ``layer_methods()`` entry, raising with the valid names if ``name`` is not one."""
    methods = layer_methods()
    if name not in methods:
        raise ValueError(f"[ERROR] Unknown layer method: {name}. Expected one of {sorted(methods)}.")
    return methods[name]

def unused_layers():
    """Layer methods no cipher module calls: available, but with no worked example."""
    return sorted(name for name, entry in layer_methods().items()
                  if not entry["primitive_modules_using"])

def documentation_gaps():
    """``{layer method: undocumented parameters}``, i.e. what a caller cannot learn from the code."""
    return {name: entry["undocumented"] for name, entry in layer_methods().items()
            if entry["undocumented"]}
