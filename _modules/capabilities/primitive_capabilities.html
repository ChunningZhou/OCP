"""The cipher catalogue: which primitives exist, how a factory is called, which versions it takes."""

import ast
import importlib
import inspect
import os


_PRIMITIVES_PACKAGE = "primitives"

# A factory's `version` argument follows the kind: a permutation takes the state bit size as a
# single number, a block cipher takes [block_size, key_size]. Two exceptions: KNOT_PERMUTATION
# defaults to None and reads primitives/knot.py's own table, and SHACAL2_BLOCKCIPHER, a block
# cipher, takes a single number.
VERSION_SINGLE_NUMBER = "single_number"
VERSION_BLOCK_AND_KEY = "block_and_key"
VERSION_ABSENT = "absent"
VERSION_FROM_TABLE = "from_module_table"
VERSION_UNKNOWN = "unknown"


def _modules(package_name, skip, include_in_development):
    """Import and return the package's modules in filename order, skipping unfinished ones.

    From the directory, not a hand-written list, which would go stale as modules are added.
    A module setting ``IN_DEVELOPMENT = True`` declares itself not ready for use.
    """
    modules = []
    package = importlib.import_module(package_name)
    for filename in sorted(os.listdir(os.path.dirname(inspect.getfile(package)))):
        if not filename.endswith(".py") or filename == "__init__.py" or filename in skip:
            continue
        module_name = f"{package_name}.{filename[:-3]}"
        try:
            module = importlib.import_module(module_name)
        except Exception as exc:
            raise RuntimeError(f"[ERROR] Cannot enumerate {package_name}: importing {module_name} failed: {exc}") from exc
        if getattr(module, "IN_DEVELOPMENT", False) and not include_in_development:
            continue
        modules.append(module)
    return modules


def _parameters(signature):
    """``{name: {required, default}}``, keeping the two reasons a default is absent apart.

    Read as ``{name: default}`` a required argument and one defaulting to ``None`` were the same
    entry, so a consumer could not tell whether a bare ``FACTORY()`` was legal. Separate because
    the distinction is in the signature, not because any factory is required today -- none is.
    """
    return {
        name: {"required": value.default is inspect.Parameter.empty,
               "default": None if value.default is inspect.Parameter.empty else value.default}
        for name, value in signature.parameters.items()
    }


def _version_shape(parameter, version_rounds):
    """The shape of the ``version`` argument, from its default and whether a table was found.

    A ``None`` default alone does not make it table-driven -- ``KNOT_PERMUTATION`` is the only
    factory that means it that way, and it is only knowable because the table is there. Without
    one, the default says nothing and the shape is ``VERSION_UNKNOWN`` rather than a guess.
    """
    if parameter is None:
        return VERSION_ABSENT
    if isinstance(parameter.default, (list, tuple)):
        return VERSION_BLOCK_AND_KEY
    if parameter.default is None:
        return VERSION_FROM_TABLE if version_rounds else VERSION_UNKNOWN
    return VERSION_SINGLE_NUMBER


def _version_keys_are_arguments(shape, version_rounds):
    """Whether the table's **keys** can be passed as ``version``; ``None`` without a table.

    ``False`` is a statement about the keys, not about the versions: three block ciphers take
    ``version=[block, key]`` but key ``full_rounds`` on one component (AES and LED on the key
    size, GIFT on the block size), so passing a key straight back raises
    ``TypeError: 'int' object is not subscriptable``. Those versions are perfectly usable -- build
    the pair yourself. ``None`` means there is no table, so the question was never asked.
    """
    if not version_rounds:
        return None
    keyed_by_pair = all(isinstance(key, tuple) for key in version_rounds)
    if shape in (VERSION_BLOCK_AND_KEY, VERSION_FROM_TABLE):
        return keyed_by_pair
    if shape == VERSION_SINGLE_NUMBER:
        return not keyed_by_pair
    return False


def _parsed_module(module):
    with open(inspect.getfile(module), encoding="utf-8") as handle:
        return ast.parse(handle.read())


def _instantiated_class(tree, factory_name):
    """The primitive class a factory builds, **statically inferred**; ``None`` when inference fails.

    Recognises one pattern: a class defined in the same module, called directly in the factory
    body. A factory that delegated construction to a helper would read as ``None``, which means
    "could not be determined", never "builds nothing". All 34 factories resolve today and a test
    keeps it that way, so a ``None`` here is a signal to widen this function.
    """
    classes = {n.name for n in tree.body if isinstance(n, ast.ClassDef)}
    fn = next((n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == factory_name), None)
    if fn is None:
        return None
    called = {n.func.id for n in ast.walk(fn)
              if isinstance(n, ast.Call) and isinstance(n.func, ast.Name) and n.func.id in classes}
    return called.pop() if len(called) == 1 else None


def _literal_dict(node, name):
    """``ast.literal_eval`` of an assignment to ``name``, or ``None``."""
    if (isinstance(node, ast.Assign) and len(node.targets) == 1
            and isinstance(node.targets[0], ast.Name) and node.targets[0].id == name
            and isinstance(node.value, ast.Dict)):
        try:
            return ast.literal_eval(node.value)
        except ValueError:
            return None
    return None

def version_tables(module):
    """``{class or factory name: {version: full round count}}`` for the multi-version primitives.

    The only place the **documented** versions are written down; building gives one round count,
    not the catalogue. Two idioms normalised to one shape: ``full_rounds`` in a class
    ``__init__``, and a module-level ``<FACTORY>_VERSIONS`` carrying ``nbr_rounds``
    (``primitives/knot.py``). A single-version primitive has no table.

    **Enforced since 2026-10-10**: each of the 13 raises
    ``[ERROR] Unknown version <v> for <Class>. Expected one of <keys>.`` before building anything,
    guarded by ``test_every_version_table_is_enforced``. Until then none of them checked, and
    ``SPECK_PERMUTATION(version=12345)`` returned ``SPECK12345_PERM`` with a 6172-bit word.
    The **absence** of a table is still not evidence that anything goes: 21 factories have none
    and nothing constrains their ``version`` -- see ``TODO.md``.
    """
    tree = _parsed_module(module)
    tables = {}
    for node in tree.body:
        for name in [t.id for t in getattr(node, "targets", []) if isinstance(t, ast.Name)]:
            if not name.endswith("_VERSIONS"):
                continue
            table = _literal_dict(node, name)
            if table and all(isinstance(v, dict) and "nbr_rounds" in v for v in table.values()):
                tables[name[: -len("_VERSIONS")]] = {k: v["nbr_rounds"] for k, v in table.items()}
    for cls in [n for n in tree.body if isinstance(n, ast.ClassDef)]:
        init = next((m for m in cls.body if isinstance(m, ast.FunctionDef) and m.name == "__init__"), None)
        for node in ast.walk(init) if init else ():
            table = _literal_dict(node, "full_rounds")
            if table:
                tables[cls.name] = table
    return tables

def primitive_factories(include_in_development=False):
    """``{(module, factory): {...}}`` for every module-level cipher factory. Builds nothing.

    Per entry: ``parameters`` as ``{name: {required, default}}``, kept apart because a required
    argument and one defaulting to ``None`` are both absent from a signature read as
    ``{name: default}`` -- today **no factory has a required argument**, so every one of the 32
    ready factories can be called with none; ``version_shape`` and ``version_default``;
    ``builds`` (statically inferred, see ``_instantiated_class``); ``version_rounds`` (the
    documented table, which the factory now enforces -- see ``version_tables``) and
    ``version_keys_are_arguments``; ``in_development``.
    """
    catalogue = {}
    for module in _modules(_PRIMITIVES_PACKAGE, ("primitives.py",), include_in_development):
        tree = _parsed_module(module)
        tables = version_tables(module)
        short_name = module.__name__.split(".")[-1]
        for name, factory in inspect.getmembers(module, inspect.isfunction):
            if factory.__module__ != module.__name__ or not name.isupper():
                continue
            signature = inspect.signature(factory)
            version = signature.parameters.get("version")
            builds = _instantiated_class(tree, name)
            version_rounds = tables.get(builds) or tables.get(name)
            shape = _version_shape(version, version_rounds)
            catalogue[(short_name, name)] = {
                "module": short_name,
                "factory": name,
                "parameters": _parameters(signature),
                "version_shape": shape,
                "version_default": None if version is None else version.default,
                "builds": builds,
                "version_rounds": version_rounds,
                "version_keys_are_arguments": _version_keys_are_arguments(shape, version_rounds),
                "in_development": bool(getattr(module, "IN_DEVELOPMENT", False)),
            }
    return catalogue

def describe_primitive(module, factory, **kwargs):
    """Build one cipher and report what only an instance can say.

    Adds ``name``, ``kind``, ``nbr_rounds``, ``test_vectors`` and the built ``cipher`` itself to
    the static entry, or ``error`` when the factory raises. Separate from ``primitive_factories``
    because building all 34 takes about 3 seconds.

    ``cipher`` is handed back so a caller that needs the instance does not build a second one --
    ``TRIVIUM`` alone costs 2.7 s, and two builds are two chances to differ. It is the one
    non-serialisable field in the entry; drop it before writing the entry out.
    """
    catalogue = primitive_factories(include_in_development=True)
    if (module, factory) not in catalogue:
        known = sorted(f for m, f in catalogue if m == module)
        if not known:
            raise ValueError(f"[ERROR] Unknown primitive module: {module}. "
                             f"Expected one of {sorted({m for m, _ in catalogue})}.")
        raise ValueError(f"[ERROR] Unknown factory {factory} in primitives/{module}.py. "
                         f"Expected one of {known}.")
    entry = dict(catalogue[(module, factory)])
    target = getattr(importlib.import_module(f"{_PRIMITIVES_PACKAGE}.{module}"), factory)
    try:
        cipher = target(**kwargs)
    except Exception as exc:
        entry["error"] = f"{type(exc).__name__}: {exc}"
        return entry
    entry.update(name=cipher.name, kind=type(cipher).__mro__[1].__name__,
                 nbr_rounds=cipher.nbr_rounds, test_vectors=len(cipher.test_vectors),
                 cipher=cipher)
    return entry
