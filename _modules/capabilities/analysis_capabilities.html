# Derived from the five vocabulary constants tools/ keeps, which is where the pipeline reads them.
# These live here instead because nothing in tools/ or attacks/ uses them: the impossible and
# zero-correlation searches each take their own two goals and resolve them inline, so only a
# consumer asking "which goals exist at all" needs the union.
IMPOSSIBLE_GOALS = tuple(IMPOSSIBLE_GOAL_MIDDLE)
ZEROCORR_GOALS = tuple(ZEROCORR_GOAL_MIDDLE)
ALL_GOALS = DIFFERENTIAL_GOALS + LINEAR_GOALS + INTEGRAL_GOALS + IMPOSSIBLE_GOALS + ZEROCORR_GOALS
_GOAL_MIDDLE = dict(IMPOSSIBLE_GOAL_MIDDLE, **ZEROCORR_GOAL_MIDDLE)

def resolve_goal(goal):
    """The goal actually modelled for ``goal``: itself, or the middle-round goal it is built on."""
    return _GOAL_MIDDLE.get(goal, goal)



# Answers come from SUPPORTED_MODEL_VERSIONS, not from what generate_model implements;
# operator_capabilities.consistency_report() is what keeps the two from drifting.
SUPPORTED = "supported"
UNSUPPORTED = "unsupported"
UNVERIFIED = "unverified"       # declares no model versions, so the check cannot answer


def _validate_goal(goal):
    """Accepts all 13 goals, including the 4 that are modelled as another one.

    ``GOAL_MODEL_VERSION_RULES`` holds the 9 goals that have model-version rules of their own. The
    impossible-differential and zero-correlation goals have none: each is modelled as its
    middle-round goal, so asking about one means asking about that. Before the vocabularies were
    joined this function rejected those 4 outright, and the capability layer answered "no such
    goal" for a goal ``impossible_differential_attacks`` takes by default.
    """
    if goal not in ALL_GOALS:
        raise ValueError(f"[ERROR] Invalid goal: {goal}. Expected one of {sorted(ALL_GOALS)}.")


def _validate_model_type(model_type):
    if model_type not in MODEL_TYPES:
        raise ValueError(f"[ERROR] Invalid model_type: {model_type}. Expected one of {list(MODEL_TYPES)}.")

def required_version(operator_class, goal):
    """The model-version suffix ``goal`` requires for ``operator_class``.

    Applies ``GOAL_MODEL_VERSION_RULES`` as ``set_model_versions`` does: in order, each matching
    rule overriding the previous. Raises when no rule covers the operator, which means the goal
    is missing its ``scope=None`` fallback.
    """
    _validate_goal(goal)
    goal = resolve_goal(goal)
    suffix = None
    for version, scope in GOAL_MODEL_VERSION_RULES[goal]:
        if operator_in_scope(operator_class, scope):
            suffix = version
    if suffix is None:
        raise ValueError(
            f"[ERROR] Goal {goal} has no model-version rule covering {operator_class.__name__}. "
            f"GOAL_MODEL_VERSION_RULES[{goal!r}] needs a (version, None) fallback rule."
        )
    return suffix

def operator_support(operator_class, goal, model_type):
    """Whether ``operator_class`` declares the version ``goal`` needs: one of the three states above."""
    _validate_model_type(model_type)
    if operator_class in NO_MODEL_OPERATORS:
        return SUPPORTED
    declared = getattr(operator_class, "SUPPORTED_MODEL_VERSIONS", None) or {}
    if not declared:
        return UNVERIFIED
    return SUPPORTED if required_version(operator_class, goal) in declared.get(model_type, ()) else UNSUPPORTED

def model_scope_constraints(cipher, config_model=None):
    """Exactly the operators ``gen_round_model_constraint_obj_fun`` would generate a model for.

    Mirrors the generation loop, honouring ``positions`` and the ``gen_input_model`` /
    ``gen_output_model`` switches; ``set_model_versions`` sweeps wider on purpose. A list, not a
    generator: callers make one full pass, and a generator would go empty on a second.
    """
    config_model = config_model or {}
    functions, rounds, layers, positions = fill_functions_rounds_layers_positions(
        cipher,
        functions=config_model.get("functions"),
        rounds=config_model.get("rounds"),
        layers=config_model.get("layers"),
        positions=config_model.get("positions"),
    )
    constraints = []
    if config_model.get("gen_input_model", True):
        constraints.extend(cipher.inputs_constraints)
    if config_model.get("gen_output_model", True):
        constraints.extend(cipher.outputs_constraints)
    for f in functions:
        for r in rounds[f]:
            for l in layers[f][r]:
                constraints.extend(cipher.functions[f].constraints[r][l][i] for i in positions[f][r][l])
    return constraints

def scope_operator_classes(cipher, config_model=None):
    """``{class name: class}`` for every distinct operator inside the query scope."""
    return {type(cons).__name__: type(cons) for cons in model_scope_constraints(cipher, config_model)}

def goal_report(cipher, goal, model_type, config_model=None, scope_classes=None):
    """A structured answer for one (goal, model_type) pair; ``capability_report`` gives all of them.

    Keys: ``feasible`` (nothing in scope declares it cannot), ``verified`` (and every operator
    declares the required version -- about declarations, not an end-to-end check), ``blockers``
    and ``unverified`` as sorted ``(class, suffix)`` lists, plus ``goal``, ``modelled_as``,
    ``model_type`` and ``operators``. Read-only: assigns no ``model_version``.

    ``modelled_as`` differs from ``goal`` for the 4 impossible-differential and zero-correlation
    goals: the answer is about the middle-round goal they are built on, which is the only thing
    the operators declare anything about.
    """
    _validate_goal(goal)
    _validate_model_type(model_type)
    classes = scope_classes if scope_classes is not None else scope_operator_classes(cipher, config_model)
    blockers, unverified = [], []
    for name, operator_class in sorted(classes.items()):
        state = operator_support(operator_class, goal, model_type)
        if state is UNSUPPORTED:
            blockers.append((name, required_version(operator_class, goal)))
        elif state is UNVERIFIED:
            unverified.append((name, required_version(operator_class, goal)))
    return {"goal": goal, "modelled_as": resolve_goal(goal), "model_type": model_type,
            "feasible": not blockers, "verified": not blockers and not unverified,
            "blockers": blockers, "unverified": unverified, "operators": len(classes)}

def capability_report(cipher, model_type, config_model=None):
    """``{goal: goal_report(...)}`` for every goal, walking the cipher's operators only once."""
    _validate_model_type(model_type)
    classes = scope_operator_classes(cipher, config_model)
    return {goal: goal_report(cipher, goal, model_type, scope_classes=classes)
            for goal in ALL_GOALS}

def feasible_goals(cipher, model_type="milp", config_model=None, require_verified=True):
    """The goals ``cipher`` can express under ``model_type``, in ``ALL_GOALS`` order.

    ``require_verified=True`` (the default) keeps only goals every operator declares. ``False``
    also accepts goals held up solely by operators that declare nothing, which are not known to
    work: the late ``version ... not existing`` error is what such an operator produces.
    """
    report = capability_report(cipher, model_type, config_model)
    key = "verified" if require_verified else "feasible"
    return [goal for goal, entry in report.items() if entry[key]]

def explain_goal(cipher, goal, model_type, config_model=None):
    """``goal_report`` rendered for a human or an error message; ``""`` when nothing is wrong."""
    report = goal_report(cipher, goal, model_type, config_model)
    parts = []
    if report["blockers"]:
        listed = ", ".join(f"{name} (needs {name}_{version})" for name, version in report["blockers"])
        parts.append(f"{len(report['blockers'])} operator(s) do not model it: {listed}")
    if report["unverified"]:
        listed = ", ".join(f"{name} (would need {name}_{version})" for name, version in report["unverified"])
        parts.append(f"{len(report['unverified'])} operator(s) declare no model versions, so the goal is "
                     f"UNVERIFIED for them: {listed}")
    if not parts:
        return ""
    middle = report["modelled_as"]
    named = goal if middle == goal else f"{goal} (modelled as {middle})"
    return (f"[ERROR] Goal {named} is not confirmed for {cipher.name} under "
            f"model_type='{model_type}': " + "; ".join(parts) + ".")

def goal_support_matrix():
    """``{goal: {model_type: {state: sorted class names}}}`` over every concrete operator class.

    Cipher-independent: what any cipher could express, not what this one can. All three states are
    reported, so an operator that declares nothing stays under ``UNVERIFIED``.
    """
    classes = operator_classes()
    matrix = {}
    for goal in ALL_GOALS:
        matrix[goal] = {}
        for model_type in MODEL_TYPES:
            by_state = {SUPPORTED: [], UNSUPPORTED: [], UNVERIFIED: []}
            for name, operator_class in sorted(classes.items()):
                by_state[operator_support(operator_class, goal, model_type)].append(name)
            matrix[goal][model_type] = by_state
    return matrix

# The three shapes attacks/attacks.py returns. Three of its six entry points return no AttackTrace,
# so `.data`, `to_dict` and every `save_*` are unavailable there -- writing `result[0].data[...]`
# against a trail result and reusing it is the most misleading thing a consumer can do here.
TRACE_LIST = "attack_trace_list"
POSITION_PAIRS = "position_pair_list"
VALUE_PAIR = "value_pair"

_TRACE_MODULE = "attacks.attack_trace"
_ATTACKS_MODULE = "attacks.attacks"
_REFERENCE_PACKAGE = "recheck"

# Where each trace class's `.data` is built. Reading the class body alone finds only the keys it
# reads back, which was 6 of the 9 a real trail carries -- a partial answer in the place the F
# axis is asked its headline question. Keyed by the class the writer instantiates, subclasses
# included; a test compares the union against a real result and fails if either side drifts.
_DATA_WRITERS = {
    "Trail": ("attacks.attack_trace", "extract_and_format_trails"),
    "IntegralDistinguisher": ("attacks.integral_cryptanalysis",
                              "_extract_and_format_integral_distinguishers"),
}

# The two goals two_stage_trail_search accepts. It is the one entry point whose goals are not a
# vocabulary family: it branches on these two names and refuses the other 11, and it spans both
# the differential and the linear family. A test probes all 13 against it to keep this honest.
TWO_STAGE_GOALS = ("DIFFERENTIALPATH_PROB", "LINEARPATH_CORR")

# Entry point -> (shape, element class, the goals it accepts). The shape is declared, because a
# return shape is not in the source in any readable form; a test runs each entry point and
# confirms it. The goals are the vocabulary partition from tools/model_configuration.py, which is
# what lets a goal be mapped to the result it produces.
_ENTRY_POINTS = {
    "diff_attacks": (TRACE_LIST, "DifferentialTrail", DIFFERENTIAL_GOALS),
    "linear_attacks": (TRACE_LIST, "LinearTrail", LINEAR_GOALS),
    "integral_attacks": (TRACE_LIST, "IntegralDistinguisher", INTEGRAL_GOALS),
    "impossible_differential_attacks": (POSITION_PAIRS, None, IMPOSSIBLE_GOALS),
    "zero_correlation_attacks": (POSITION_PAIRS, None, ZEROCORR_GOALS),
    "two_stage_trail_search": (VALUE_PAIR, None, TWO_STAGE_GOALS),
}

# What an empty or None result means, and what it cannot be told apart from. The repository's rule
# is that a masked failure stays distinguishable from a genuine absence; these are the places a
# consumer has to apply it, because the value itself carries no such distinction.
_EMPTY_MEANING = {
    TRACE_LIST: ("[] means no solution at the bound the model was given. A backend that cannot "
                 "load raises, and a non-list from the solver raises, so [] is not a failure -- "
                 "but it does not say whether the objective was empty or the bound was capped."),
    POSITION_PAIRS: ("[] means no position pair among those enumerated was confirmed impossible. "
                     "The enumeration is bounded by min/max_active_in/out, so [] over a narrow "
                     "range says nothing about pairs outside it."),
    VALUE_PAIR: ("best_weight is None when stage 2 found no trail, which is indistinguishable "
                 "from stage 2 never running: stage 1 returning no active-S-box count leaves "
                 "the same None."),
}


def _subscript_key(node):
    """The string a subscript indexes with, or ``None``.

    Python 3.8 wraps it in ``ast.Index`` while 3.9+ puts the ``Constant`` there directly. Missing
    that reads every class as carrying no data keys -- an empty answer that looks plausible.
    """
    if isinstance(node, ast.Index):
        node = node.value
    value = getattr(node, "value", None)
    return value if isinstance(value, str) else None


def _data_keys_in(class_node):
    """The ``self.data`` keys a class body reads, by the three idioms it uses to read them."""
    keys = set()
    for node in ast.walk(class_node):
        if isinstance(node, ast.Subscript) and isinstance(node.value, ast.Attribute) \
                and node.value.attr == "data":
            key = _subscript_key(node.slice)
            if key is not None:
                keys.add(key)
        elif isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute) \
                and node.func.attr == "get" and isinstance(node.func.value, ast.Attribute) \
                and node.func.value.attr == "data" and node.args:
            key = getattr(node.args[0], "value", None)
            if isinstance(key, str):
                keys.add(key)
        elif isinstance(node, ast.Compare) and isinstance(node.left, ast.Constant) \
                and any(isinstance(op, ast.In) for op in node.ops) \
                and any(isinstance(c, ast.Attribute) and c.attr == "data" for c in node.comparators):
            if isinstance(node.left.value, str):
                keys.add(node.left.value)
    return keys


def _constructed_keys(module_name, function_name):
    """``(constant keys, unresolved keys)`` of the ``data = {...}`` a writer builds.

    Two of the trail keys are variables (``weight_key``, ``rounds_weight_key``) because the writer
    is shared by the differential and linear classes. Those cannot be resolved statically, so they
    are reported rather than dropped -- the class body reads them back, so the union still covers
    them.
    """
    module = importlib.import_module(module_name)
    tree = ast.parse(inspect.getsource(module))
    function = next((n for n in ast.walk(tree)
                     if isinstance(n, ast.FunctionDef) and n.name == function_name), None)
    if function is None:
        raise ValueError(f"[ERROR] {module_name} has no function {function_name}, so the keys of "
                         f"the .data it builds cannot be read. Update _DATA_WRITERS.")
    for node in ast.walk(function):
        if isinstance(node, ast.Assign) and len(node.targets) == 1 \
                and getattr(node.targets[0], "id", None) == "data" \
                and isinstance(node.value, ast.Dict):
            constants = {k.value for k in node.value.keys
                         if isinstance(k, ast.Constant) and isinstance(k.value, str)}
            unresolved = sorted(getattr(k, "id", "?") for k in node.value.keys
                                if not isinstance(k, ast.Constant))
            return constants, unresolved
    raise ValueError(f"[ERROR] {module_name}.{function_name} no longer builds a `data = {{...}}` "
                     f"literal, so its keys cannot be read. Update _DATA_WRITERS.")


def _exports_in(class_node):
    """``{format: state}`` for each ``save_*``: implemented, a stub that raises, or abstract."""
    exports = {}
    for method in class_node.body:
        if not (isinstance(method, ast.FunctionDef) and method.name.startswith("save_")):
            continue
        if any(getattr(d, "id", "") == "abstractmethod" for d in method.decorator_list):
            exports[method.name[len("save_"):]] = "abstract"
            continue
        body = [s for s in method.body
                if not (isinstance(s, ast.Expr) and isinstance(s.value, ast.Constant))]
        raises = [s for s in body if isinstance(s, ast.Raise)
                  and getattr(getattr(s.exc, "func", None), "id", "") == "NotImplementedError"]
        exports[method.name[len("save_"):]] = "unimplemented" if raises and len(raises) == len(body) \
            else "implemented"
    return exports

def trace_classes():
    """``{class name: {...}}`` for every ``AttackTrace`` class, merged along the real MRO.

    Per entry: ``bases``; ``data_keys``, every key an instance carries; ``read_keys``, the subset
    the class bodies read back; ``constructed_keys``, the subset its writer sets;
    ``unresolved_constructed``, writer keys that are variables and so not statically known; and
    ``exports`` as ``{format: implemented | unimplemented | abstract}``. All read out of the
    source, so none of it can disagree with the code.
    """
    module = importlib.import_module(_TRACE_MODULE)
    tree = ast.parse(inspect.getsource(module))
    nodes = {n.name: n for n in tree.body if isinstance(n, ast.ClassDef)}
    own_keys = {name: _data_keys_in(node) for name, node in nodes.items()}
    own_exports = {name: _exports_in(node) for name, node in nodes.items()}
    report = {}
    for name, node in nodes.items():
        chain = [c.__name__ for c in getattr(module, name).__mro__ if c.__name__ in nodes]
        read, exports = set(), {}
        for ancestor in reversed(chain):          # base first, so a subclass overrides it
            read |= own_keys[ancestor]
            exports.update(own_exports[ancestor])
        written, unresolved = set(), []
        for served, writer in _DATA_WRITERS.items():
            if served in chain:
                written, unresolved = _constructed_keys(*writer)
        report[name] = {
            "bases": [b.id for b in node.bases if isinstance(b, ast.Name)],
            "data_keys": sorted(read | written),
            "read_keys": sorted(read),
            "constructed_keys": sorted(written),
            "unresolved_constructed": unresolved,
            "own_data_keys": sorted(own_keys[name]),
            "exports": exports,
        }
    return report

def export_formats(class_name=None):
    """``{format: state}`` for one trace class, or the union over the concrete ones.

    ``unimplemented`` is a method that exists and raises ``NotImplementedError``: having a
    ``save_tex`` is not being able to export LaTeX. ``mixed`` would mean two concrete classes
    disagree, which none do today.
    """
    classes = trace_classes()
    if class_name is None:
        merged = {}
        for entry in classes.values():
            for name, state in entry["exports"].items():
                if state == "abstract":           # the ABC declares, it does not answer
                    continue
                merged[name] = state if merged.get(name) in (None, state) else "mixed"
        return merged
    if class_name not in classes:
        raise ValueError(f"[ERROR] Unknown attack-trace class: {class_name}. "
                         f"Expected one of {sorted(classes)}.")
    return classes[class_name]["exports"]

def recheck_available():
    """What ``reference/`` offers for independently re-checking a result.

    ``reference`` is an **optional** dependency -- published results, not part of the pipeline --
    so its absence is reported rather than raised. ``analyses`` comes from ``reference.CLAIMS``:
    an analysis missing from that vocabulary cannot be recorded as a claim, let alone replayed.
    """
    try:
        reference = importlib.import_module(_REFERENCE_PACKAGE)
    except Exception as exc:
        return {"available": False, "reason": f"{type(exc).__name__}: {exc}",
                "analyses": [], "checks": []}
    return {"available": True, "reason": "",
            "analyses": sorted({claim[0] for claim in reference.CLAIMS}),
            "checks": sorted(reference.CHECKS)}

def recheck_parameters():
    """``{check: {name: {required, default}}}`` -- what a replay has to be given.

    ``recheck_path`` says which checks apply and never what they need. Read from each check's own
    signature, so the four turn out to take the same four arguments: the claim ``spec``, an
    optional pre-built ``cipher``, a ``model_type`` and a ``config_solver``. The round count and
    the bound are not arguments -- they come out of the claim, which is why a replay needs the
    claim file and not a configuration of its own.
    """
    report = recheck_available()
    if not report["available"]:
        return {}
    reference = importlib.import_module(_REFERENCE_PACKAGE)
    parameters = {}
    # reference.CHECKS maps the name to the function, which is more reliable than guessing
    # `check_<name>`: those four are not re-exported from the package's __all__.
    for check, function in reference.CHECKS.items():
        if check not in report["checks"] or not callable(function):
            continue
        parameters[check] = {
            name: {"required": value.default is inspect.Parameter.empty,
                   "default": None if value.default is inspect.Parameter.empty else value.default}
            for name, value in inspect.signature(function).parameters.items()
        }
    return parameters

def recheck_path(entry_point):
    """``(applicable checks, reason)`` for independently re-checking what ``entry_point`` returns.

    Both the analysis and the shape matter. ``reference``'s four checks replay a **trail** -- pin
    it into the model, solve, compare -- so a result that is not a trail cannot go through them
    whatever its analysis. A bare optimum still has a path: ``reference.compare_bound``.
    """
    if entry_point not in _ENTRY_POINTS:
        raise ValueError(f"[ERROR] Unknown analysis entry point: {entry_point}. "
                         f"Expected one of {sorted(_ENTRY_POINTS)}.")
    shape, _, goals = _ENTRY_POINTS[entry_point]
    # reference/ keys its claims by analysis, so map the goal set onto one. two_stage spans both
    # families; its number is compared the same way either way, so differential stands for it.
    analysis = "integral" if goals == INTEGRAL_GOALS else \
               "linear" if goals in (LINEAR_GOALS, ZEROCORR_GOALS) else "differential"
    report = recheck_available()

    # Permanent reasons first, then the environmental one. A result that is not a trail stays
    # unreplayable however this checkout is set up, so answering "reference/ is missing" there
    # would send the reader to install something that would not help.
    if shape == POSITION_PAIRS:
        return [], ("Position pairs are not a trail, and the four checks replay a trail. No "
                    "reference/ path takes them whatever is installed, so an "
                    "impossible-differential or zero-correlation result stands on this "
                    "repository alone.")
    if shape == VALUE_PAIR:
        hint = (" reference.compare_bound(spec, rounds, got) tests the number against a claim."
                if report["available"] else "")
        return [], ("The four checks replay a trail, and this returns a number." + hint)
    if not report["available"]:
        # Not a failure: reference/ holds published results and main does not track it.
        return [], (f"Re-checking is unavailable in this checkout: the optional reference/ "
                    f"package is not importable ({report['reason']}).")
    if analysis not in report["analyses"]:
        return [], (f"reference/ has no claim vocabulary for {analysis}, only "
                    f"{report['analyses']}. A result stands on this repository alone.")
    return report["checks"], ""

def entry_point_parameters(entry_point):
    """``{name: {required, default}}`` for what an entry point takes, from its own signature.

    ``result_shapes`` said what comes back and never what goes in, so a consumer could read the
    goal and the objective-target vocabularies without learning which argument takes which --
    or that ``two_stage_trail_search`` is handed a **factory and a round count** where the other
    five take a built cipher. Same shape as the A axis reports a factory's parameters, so the
    two read alike.
    """
    if entry_point not in _ENTRY_POINTS:
        raise ValueError(f"[ERROR] Unknown analysis entry point: {entry_point}. "
                         f"Expected one of {sorted(_ENTRY_POINTS)}.")
    function = getattr(importlib.import_module(_ATTACKS_MODULE), entry_point)
    return {name: {"required": value.default is inspect.Parameter.empty,
                   "default": None if value.default is inspect.Parameter.empty else value.default}
            for name, value in inspect.signature(function).parameters.items()}

def result_shapes():
    """``{entry point: {...}}`` for the six entry points of ``attacks/attacks.py``.

    Per entry: ``parameters``, what the call takes (see ``entry_point_parameters``);
    ``returns`` (one of the three shape constants); ``element`` with the ``data_keys``
    and ``exports`` that come with it, empty when the shape carries no ``AttackTrace``; ``goals``,
    the goals this entry point accepts; ``recheck_checks`` with ``recheck`` giving the reason when
    there are none; and ``empty_means``, what an empty result does **not** tell you.
    """
    classes = trace_classes()
    report = {}
    for name, (shape, element, goals) in _ENTRY_POINTS.items():
        entry = classes.get(element) or {}
        checks, reason = recheck_path(name)
        report[name] = {
            "parameters": entry_point_parameters(name),
            "returns": shape,
            "element": element,
            "data_keys": entry.get("data_keys", []),
            "exports": entry.get("exports", {}),
            "goals": list(goals),
            "recheck_checks": checks,
            "recheck_parameters": {c: p for c, p in recheck_parameters().items() if c in checks},
            "recheck": reason,
            "empty_means": _EMPTY_MEANING[shape],
        }
    return report

def result_for_goal(goal):
    """What running ``goal`` gives back: its entry point's ``result_shapes()`` entry.

    The join the two halves of this module exist to make: ``feasible_goals`` says a goal can be
    expressed, this says what comes back and whether it can be re-checked. Several entry points
    can accept one goal -- ``DIFFERENTIAL_GOALS`` reaches both ``diff_attacks`` and
    ``two_stage_trail_search`` -- so the answer is a dict keyed by entry point.
    """
    _validate_goal(goal)
    shapes = result_shapes()
    return {name: shapes[name] for name, entry in _ENTRY_POINTS.items() if goal in entry[2]}

def describe_result(entry_point):
    """One ``result_shapes()`` entry, raising with the valid names if there is no such entry point."""
    shapes = result_shapes()
    if entry_point not in shapes:
        raise ValueError(f"[ERROR] Unknown analysis entry point: {entry_point}. "
                         f"Expected one of {sorted(shapes)}.")
    return shapes[entry_point]

def solver_backends():
    """Which solver backends are installed **on this host**, per model type.

    Host state, not a repository fact: the same checkout answers differently elsewhere, so this
    never belongs in a cached manifest without the host it came from. A goal being feasible says
    the model can be built, not that anything here can solve it -- a backend that cannot load
    raises rather than returning an empty result, which is deliberate.

    ``TODO.md`` records the known limit: a backend whose module imports but whose native library

    solving = importlib.import_module("solving.solving")
    return solving.solver_capabilities()

def shape_asymmetry():
    """The two things these signatures hide, which no other function here states outright."""
    carriers = sorted(n for n, e in _ENTRY_POINTS.items() if e[0] == TRACE_LIST)
    others = sorted(n for n, e in _ENTRY_POINTS.items() if e[0] != TRACE_LIST)
    return (f"Only {len(carriers)} of {len(_ENTRY_POINTS)} entry points return AttackTrace "
            f"objects ({', '.join(carriers)}). {', '.join(others)} return plain tuples, so "
            f".data, to_dict() and every save_* are unavailable there -- code written against a "
            f"trail result breaks on half the analyses. And two_stage_trail_search takes "
            f"(cipher_factory, r), not a built cipher like the other five, because it builds one "
