"""One place to ask what OCP can do, composed from the per-axis modules beside this file."""

import importlib

from capabilities import (analysis_capabilities, implementation_capabilities, layer_capabilities,
                          operator_capabilities, primitive_capabilities)

# Nothing under attacks/, implementations/ or solving/ imports this package: the pipeline does not
# need to be told what it can do. The consumers are an LLM working in the repository, OCP-agent's
# manifest, and the documentation. The dependency runs facade -> axis, never the other way, so no
# axis module imports this one.
AXES = {
    "primitives": primitive_capabilities,
    "layer_methods": layer_capabilities,
    "operators": operator_capabilities,
    "analysis": analysis_capabilities,
    "implementations": implementation_capabilities,
}


def _analysis_summary(report):
    """``{feasible, blocked, unverified}`` over a ``capability_report`` -- three groups, not two.

    A goal held up only by operators that declare nothing is neither feasible nor blocked, and a
    two-way summary left it out of both: a consumer could not tell "cannot" from "nobody looked".
    Separate from ``describe_cipher`` so the partition can be checked on a report that *has* an
    unverified goal -- today's operator set has none, so the real reports cannot exercise it.
    """
    return {
        "feasible": [goal for goal, e in report.items() if e["verified"]],
        "blocked": {goal: e["blockers"] for goal, e in report.items() if e["blockers"]},
        "unverified": {goal: e["unverified"] for goal, e in report.items()
                       if e["unverified"] and not e["blockers"]},
    }

def describe_cipher(module, factory, model_type="milp", **kwargs):
    """Everything the axes know about one cipher, built once and asked once.

    Joins the catalogue entry, the goals it can express under ``model_type``, and how its
    implementation can be emitted. ``kwargs`` go to the factory (``r=`` for the round count);
    about 0.2 s, almost all of it walking the operators.

    ``feasible`` means every operator **declares** the required model version, which is not an
    end-to-end check: see ``analysis_capabilities.feasible_goals``. ``unverified`` is the third
    group, goals held up only by operators that declare nothing.
    """
    entry = primitive_capabilities.describe_primitive(module, factory, **kwargs)
    if "error" in entry:
        return {"primitive": entry, "analysis": None, "implementation": None}

    cipher = entry.pop("cipher")                  # built by describe_primitive, reused here
    report = analysis_capabilities.capability_report(cipher, model_type)
    return {
        "primitive": entry,
        "analysis": dict(_analysis_summary(report), model_type=model_type, report=report),
        "implementation": {
            "languages": implementation_capabilities.feasible_languages(cipher),
            "with_automated_kat": implementation_capabilities.feasible_languages(cipher, require_kat=True),
            "operator_languages": implementation_capabilities.cipher_languages(cipher),
        },
    }

def describe_repository():
    """The static picture: what exists, without building a single cipher.

    The answer to "what is this repository" for a consumer that has just arrived. Every count
    comes from its axis, so none of them can disagree with the code.
    """
    factories = primitive_capabilities.primitive_factories()
    operators = operator_capabilities.operator_classes()
    return {
        "primitives": {
            "factories": len(factories),
            "with_version_table": sum(1 for e in factories.values() if e["version_rounds"]),
            "in_development": sorted(f for (_, f) in
                                     primitive_capabilities.primitive_factories(include_in_development=True)
                                     if (_, f) not in factories),
        },
        # "layer_methods", not "layers": these are Python methods on Layered_Function, which a
        # caller invokes directly. OCP-agent's manifest has a "layers" of its own holding the
        # declarative spec vocabulary (add_constant, subcell_sbox, ...) -- near-identical names,
        # different things, and a consumer that conflates them writes code against the wrong one.
        "layer_methods": {
            "methods": sorted(layer_capabilities.layer_methods()),
            "unused": layer_capabilities.unused_layers(),
            "undocumented_arguments": layer_capabilities.documentation_gaps(),
        },
        "operators": {
            "classes": len(operators),
            "declaring": len(operator_capabilities.operator_classes(declaring_only=True)),
            "declarations_match_source": not operator_capabilities.consistency_report(),
            "verification_coverage": operator_capabilities.verification_coverage(),
        },
        "analysis": {
            "goals": sorted(analysis_capabilities.ALL_GOALS),
            "model_types": list(analysis_capabilities.MODEL_TYPES),
            "support_matrix": analysis_capabilities.goal_support_matrix(),
            "entry_points": analysis_capabilities.result_shapes(),
            "recheck": analysis_capabilities.recheck_available(),
        },
        "implementations": {
            "languages": implementation_capabilities.languages(),
            "feasible": implementation_capabilities.feasible_languages(),
            "artifact_extensions": implementation_capabilities.artifact_extensions(),
        },
    }

def explain(cipher, goal=None, model_type="milp", language=None):
    """Why something is unavailable for ``cipher``, in one call; ``""`` when it is available.

    Delegates to the axis that owns the question -- ``goal`` to the analysis axis, ``language`` to
    the implementation axis -- so the message a consumer sees is the one the axis would give.
    ``cipher`` reaches both: asked without it, the implementation axis answers at repository scope
    and would blame a missing toolchain for a language the cipher's operators cannot emit at all.
    """
    messages = []
    if goal is not None:
        messages.append(analysis_capabilities.explain_goal(cipher, goal, model_type))
    if language is not None:
        messages.append(implementation_capabilities.explain_language(language, cipher))
    return " ".join(message for message in messages if message)

def scope():
    """What the axes cover and what they deliberately do not, with where the rest lives.

    ``axes`` answers "which axes exist". This answers "what is in the subject matter and what is
    outside it" -- a different question, and the one a consumer gets wrong, because a description
    that lists only what it knows reads as complete.

    The case that prompted it: a manifest carrying ``layer_methods`` and no statement of scope let
    a reader conclude that writing Python is the only way to describe a cipher to OCP. The note on
    ``layer_methods`` did mention OCP-agent's vocabulary, but phrased as a disambiguation between
    two similar names rather than as "there is a second route, and it is over there".

    ``covers`` is derived from ``AXES``, so a new axis cannot appear without appearing here.
    """
    return {
        "covers": sorted(AXES),
        "does_not_cover": [dict(entry) for entry in OUT_OF_SCOPE],
    }

def to_manifest():
    """Everything the axes know, in a form that survives ``json.dumps`` unchanged.

    ``describe_repository()`` is already serialisable, but two axis functions are not: the
    catalogue is keyed by the ``(module, factory)`` tuple and a block cipher's ``version_rounds``
    by ``(block_size, key_size)``, and JSON has no tuple keys. Both become lists of records here,
    so a consumer never has to guess how a tuple was flattened.

    ``host`` holds the answers that are true of this machine rather than of the repository --
    installed solvers, toolchains on PATH, whether the optional ``reference/`` package is here.
    Caching those as repository facts is how a manifest comes to describe a machine nobody has.
    """
    factories = primitive_capabilities.primitive_factories(include_in_development=True)
    catalogue = []
    for (module, factory), entry in sorted(factories.items()):
        record = dict(entry)
        table = record.pop("version_rounds")
        record["version_rounds"] = None if not table else [
            {"version": list(key) if isinstance(key, tuple) else key, "rounds": rounds}
            for key, rounds in sorted(table.items(), key=lambda kv: str(kv[0]))
        ]
        catalogue.append(record)
    return {
        "schema_version": MANIFEST_SCHEMA_VERSION,
        # `axes` stays a plain list: which axes exist and what the subject matter is are two
        # questions, and `scope` is an added field, so no schema bump.
        "axes": sorted(AXES),
        "scope": scope(),
        "primitives": catalogue,
        "layer_methods": {
            # Spelled out here because a manifest travels without the docs that explain it.
            "note": "Python methods on Layered_Function, called as cipher.functions[f].SboxLayer"
                    "(...). OCP-agent's manifest has its own 'layers' holding spec layer_type "
                    "names (add_constant, subcell_sbox, ...); the two vocabularies are separate.",
            "methods": {name: entry for name, entry in layer_capabilities.layer_methods().items()},
        },
        "operators": operator_capabilities.describe_operators(),
        "analysis": {
            "goals": sorted(analysis_capabilities.ALL_GOALS),
            "goal_middle": {goal: analysis_capabilities.resolve_goal(goal)
                            for goal in analysis_capabilities.ALL_GOALS},
            "model_types": list(analysis_capabilities.MODEL_TYPES),
            "call_vocabulary": analysis_capabilities.call_vocabulary(),
            "entry_points": analysis_capabilities.result_shapes(),
            "support_matrix": analysis_capabilities.goal_support_matrix(),
        },
        "implementations": {
            "languages": implementation_capabilities.languages(),
            "source_extensions": implementation_capabilities.source_extensions(),
            "artifact_extensions": list(implementation_capabilities.artifact_extensions()),
        },
        "host": {
            "solver_backends": analysis_capabilities.solver_backends(),
            "recheck": analysis_capabilities.recheck_available(),
            "host_dependent": analysis_capabilities.host_dependent_fields(),
        },
    }
