Source code for recheck.claim

r"""Reading a published-claim file and turning it into the objects OCP works with.

A claim file is a compact, human-readable JSON that is also a valid shell of an OCP trail JSON:
provenance and the per-round values live under ``data``, so the file can be read line by line
against the paper it came from. From there one solution feeds both consumers:

    assignments  ->  {var_id: 0/1}  ->  trail_struct  ->  Trail
                                    \->  model constraints

That split is the point of the package: the same pinned values that render as a trail for a human
to compare against the table are what constrain the model when the solver replays the claim.
"""

from importlib import import_module
from pathlib import Path
import json

from attacks.attack_trace import DifferentialTrail, LinearTrail, extract_trail_structures
from tools.model_constraints import gen_predefined_constraints, state_bits, state_var_ids
from tools.paths import get_files_dir


# (analysis, model, metric) -> (OCP goal, trail class, the data key holding the weight).
# ``model`` is None when the metric does not distinguish a single characteristic from the hull.
CLAIMS = {
    ("differential", "characteristic", "weight"):      ("DIFFERENTIALPATH_PROB", DifferentialTrail, "diff_weight"),
    ("differential", "differential", "weight"):        ("DIFFERENTIAL_PROB", DifferentialTrail, "diff_weight"),
    ("differential", None, "sbox_count"):              ("DIFFERENTIAL_SBOXCOUNT", DifferentialTrail, "diff_weight"),
    ("linear", "characteristic", "correlation"):       ("LINEARPATH_CORR", LinearTrail, "linear_weight"),
    ("linear", "hull", "correlation"):                 ("LINEARHULL_CORR", LinearTrail, "linear_weight"),
    ("linear", None, "sbox_count"):                    ("LINEAR_SBOXCOUNT", LinearTrail, "linear_weight"),
}

# The per-round series OCP reports alongside each total.
ROUND_KEY = {"diff_weight": "rounds_diff_weight", "linear_weight": "rounds_linear_weight"}

_PRIMITIVES_PACKAGE = "primitives"

# Resolved off __file__ rather than the CWD, so they mean the same thing whether an entry point
# runs as `python -m recheck.run` or `python recheck/run.py`.
#
# The claims are **not** in this package: `reference/` is the source-material store (papers, the
# claims read out of them, and the format authority EXTRACTION.md), and this package is the code
# that replays them. Keeping the two apart means the material can be present without the checker
# and the other way round, which `capabilities.recheck_available()` reports separately.
DATA_DIR = Path(__file__).resolve().parents[1] / "reference" / "data"
# The log is a run artifact, so it lands in files/ like every other one -- see AGENTS.md
# "Generated artifacts". It is read back by the table generator, which is why it is named here
# rather than inside the runner.
RESULTS_LOG = get_files_dir("recheck") / "results.jsonl"

SCHEMA_VERSION = 1

# How far a claim has been checked. A consumer that cannot tell an unread draft from a checked one
# will cite it as if it were the paper, so this is a field rather than prose buried in ``via``.
CLAIM_STATUS = {
    "unverified": "extracted from a source but not checked against the paper by a person",
    "verified": "a person checked it against the paper; ``verified_by`` says who",
    "disputed": "a known discrepancy -- see ``caveat`` before using it",
    "self": "not a published claim at all: a value this project measured, kept as a baseline",
}

def load(path):
    """Read a claim file and check the bits this package relies on are present.

    ``source`` must be an object carrying a ``status`` from ``CLAIM_STATUS``: the key merely
    existing used to be enough, so 17 files passed with ``source: null`` -- values this project
    measured itself, indistinguishable here from a published claim.
    """
    with open(path, encoding="utf-8") as f:
        spec = json.load(f)
    version = spec.get("schema_version", SCHEMA_VERSION)
    if version != SCHEMA_VERSION:
        raise ValueError(f"{path}: schema_version {version}, expected {SCHEMA_VERSION}.")
    data = spec.get("data")
    if not isinstance(data, dict):
        raise ValueError(f"{path}: no 'data' object.")
    for key in ("source", "target", "claim"):
        if key not in data:
            raise ValueError(f"{path}: data is missing {key!r}.")
    source = data["source"]
    if not isinstance(source, dict):
        raise ValueError(f"{path}: source is {source!r}, expected an object with a status.")
    status = source.get("status")
    if status not in CLAIM_STATUS:
        raise ValueError(f"{path}: source.status is {status!r}. "
                         f"Expected one of {sorted(CLAIM_STATUS)}.")
    if status == "verified" and not source.get("verified_by"):
        raise ValueError(f"{path}: status is 'verified' but verified_by is "
                         f"{source.get('verified_by')!r} -- say who checked it.")
    return spec

def resolve_claim(spec):
    """(goal, trail class, weight key) for the spec's claim."""
    claim = spec["data"]["claim"]
    key = (claim.get("analysis"), claim.get("model"), claim.get("metric"))
    if key not in CLAIMS:
        raise ValueError(f"Unsupported claim {key}. Known: {sorted(CLAIMS)}.")
    return CLAIMS[key]

def build_cipher(spec, rounds=None):
    """Instantiate the primitive the claim is about.

    A trail claim fixes the round count in ``target.kwargs``. A bound claim states a value per
    round instead, so the count is passed in here and injected under ``target.rounds_kwarg``
    (``"r"`` unless the primitive names it otherwise).
    """
    target = spec["data"]["target"]
    # The only import in this repository whose target comes from a data file. All 197 claims name
    # a `primitives.*` module, but nothing made them: an unchecked `import_module` of whatever a
    # file says is a wide door for a corpus that is 155 unreviewed drafts and will later be
    # written by an agent.
    if not str(target.get("module", "")).startswith(f"{_PRIMITIVES_PACKAGE}."):
        raise ValueError(f"target.module is {target.get('module')!r}; a claim may only name a "
                         f"module under {_PRIMITIVES_PACKAGE}/.")
    module = import_module(target["module"])
    try:
        constructor = getattr(module, target["constructor"])
    except AttributeError as exc:
        raise ValueError(f"{target['module']} has no {target['constructor']!r}.") from exc
    kwargs = dict(target.get("kwargs", {}))
    if rounds is not None:
        kwargs[target.get("rounds_kwarg", "r")] = rounds
    return constructor(**kwargs)

def assignments_to_solution(cipher, assignments):
    """``[{function, round, layer, value, nwords?}]`` -> ``{var_id: 0/1}``.

    ``value`` is the whole state as ``"0x..."``/``"0b..."``, MSB first. ``nwords`` pins only the
    first n words, for a target narrower than the state the paper wrote the value against.

    The word count and width come from the cipher, never from the claim file, so a claim cannot
    disagree with the primitive about how a state splits; a value wider than its target is
    rejected rather than truncated.
    """
    solution = {}
    for i, a in enumerate(assignments):
        where = f"assignment[{i}] ({a.get('function')} round {a.get('round')} layer {a.get('layer')})"
        try:
            fn = cipher.functions[a["function"]]
            cons_vars = fn.vars[a["round"]][a["layer"]]
        except (KeyError, IndexError) as exc:
            raise ValueError(f"{where}: no such function/round/layer.") from exc
        cons_vars = cons_vars[: a.get("nwords") or fn.nbr_words]
        # The traversal and the bit splitting live in tools; this used to be a second copy of
        # the loop in tools/model_constraints._fix_vars_to_bits.
        solution.update(dict(state_bits(cons_vars, a["value"], where)))
    return solution

def solution_to_assignments(cipher, solution, places):
    """``[{function, round, layer, value}]`` for the states ``places`` names, read off a solution.

    The inverse of ``assignments_to_solution``, for writing a claim rather than replaying one.
    ``places`` is a list of ``(function, round, layer)`` because a solution covers every variable
    in the model while a claim records a chosen few -- the 35 trail claims here hold between 6 and
    34, typically each round's input state plus the subkeys. Both directions read the state in
    ``state_var_ids`` order, so a claim written here replays unchanged.

    Whole states only: ``assignments_to_solution`` takes an ``nwords`` to pin a prefix and
    ``places`` cannot say that, so a narrowed claim is not writable from this side. No claim in
    the corpus narrows for real -- the six that carry ``nwords`` set it to the state's own width
    -- and a test holds that, since otherwise this would raise about a missing value instead.
    """
    assignments = []
    for function, rnd, layer in places:
        where = f"place ({function} round {rnd} layer {layer})"
        try:
            fn = cipher.functions[function]
            cons_vars = fn.vars[rnd][layer][: fn.nbr_words]
        except (KeyError, IndexError) as exc:
            raise ValueError(f"{where}: no such function/round/layer.") from exc
        ids = state_var_ids(cons_vars)
        missing = [var_id for var_id in ids if var_id not in solution]
        if missing:
            raise ValueError(f"{where}: the solution has no value for {missing[:4]}"
                             f"{' and others' if len(missing) > 4 else ''}.")
        bits = "".join(str(int(round(float(solution[var_id])))) for var_id in ids)
        if any(bit not in "01" for bit in bits):
            raise ValueError(f"{where}: the solution holds values other than 0/1.")
        # Hex while the width allows it exactly, since that is how the corpus is written; binary
        # otherwise, rather than padding to a nibble and changing the width.
        value = (f"0x{int(bits, 2):0{len(bits) // 4}x}" if len(bits) % 4 == 0 else f"0b{bits}")
        assignments.append({"function": function, "round": rnd, "layer": layer, "value": value})
    return assignments

def to_trail(spec, cipher, solution=None):
    """Materialize the claim as an OCP Trail, so it prints and compares like a found one."""
    goal, trail_class, weight_key = resolve_claim(spec)
    data = dict(spec["data"])
    solution = assignments_to_solution(cipher, data.get("assignments", [])) if solution is None else solution
    data["trail_struct"] = extract_trail_structures(cipher, goal, solution, "TRUNCATED")
    data.setdefault("cipher", cipher.name)
    data.setdefault(weight_key, data["claim"].get("value"))
    return trail_class(data=data, solution_trace=solution)

def pin_constraints(solution, model_type="milp"):
    """Constrain every variable the claim pins to the value it pins it to."""
    ones = [k for k, v in solution.items() if int(round(float(v))) == 1]
    zeros = [k for k, v in solution.items() if int(round(float(v))) == 0]
    return (gen_predefined_constraints(model_type, "EXACTLY", ones, 1, bitwise=False)
            + gen_predefined_constraints(model_type, "EXACTLY", zeros, 0, bitwise=False))
