    sys.path.insert(0, str(_ROOT))

PATTERNS = [str(DATA_DIR / "speck" / "liu2019_t10_*_trail.json"),
            str(DATA_DIR / "speck" / "liu2019_t12_*_trail.json")]
# PATTERNS = [str(PINNED_CLAIM)]   # one claim
# PATTERNS = [data_glob()]         # every cipher in the corpus
MODEL_TYPE = "milp"              # "milp" or "sat"
SOLVER = None                    # backend name, e.g. "GUROBI" or "Cadical"; None picks the default
TIME_LIMIT = None                # seconds per search; one that exceeds it is recorded as a failure
MAX_VERSION = None               # skip state widths above this, e.g. 64
MAX_ROUNDS = None                # skip round counts above this: a bound table is one search PER
                                 # round, so this is the difference between seconds and an evening
ANALYSES = ()                    # ("differential",) / ("linear",) to restrict; empty means both
LOG = RESULTS_LOG                # appended to, and resumed from: recorded jobs are not re-run
DRY_RUN = False                  # True lists the jobs and stops, without touching a solver

# Rough cost order. A search's difficulty is driven by the state width and the round count, so
# ordering by (width, rounds) means an interrupted run has still answered everything small.
def cost(width, rounds):
    return (width or 0) * max(rounds, 1), width or 0, rounds

def width_of(spec):
    """The primitive's state width in bits.

    Usually the ``version`` kwarg says it, but a primitive fixed to one block size takes no such
    kwarg -- PRESENT is 64 bits and says so nowhere in the claim. Falling back to 0 there would
    order full-round PRESENT ahead of every two-round SPECK32, so the width is measured off a
    one-round instance instead and cached per target.
    """
    target = spec["data"]["target"]
    version = target.get("kwargs", {}).get("version")
    if isinstance(version, list):              # a block cipher names (block, key)
        return version[0]
    if version is not None:
        return version
    key = (target["module"], target["constructor"])
    if key not in _WIDTHS:
        cipher = build_cipher(spec, rounds=1)
        function = cipher.functions[next(iter(cipher.functions))]
        # nbr_words counts the state; a round's variable list also holds temporary words
        # (LBlock keeps 32 of them beside its 64 state bits), which are not state width.
        _WIDTHS[key] = function.nbr_words * function.word_bitsize
    return _WIDTHS[key]

def label_of(spec):
    """A short name for the target, for the dry-run listing."""
    target = spec["data"]["target"]
    constructor = target["constructor"].replace("_PERMUTATION", "").replace("_BLOCKCIPHER", "")
    version = target.get("kwargs", {}).get("version")
    if isinstance(version, list):
        version = version[0]
    return constructor if version is None else f"{constructor}-{version}"

def canonical(path):
    """One identity per claim file, whatever spelling reached us.

    The resume key is the path, so `reference/data/x.json` and its absolute form have to be the
    same job -- otherwise narrowing a run to one file by typing a relative path re-runs everything
    the default (absolute) pattern already recorded. ``realpath`` matches the ``Path.resolve()``
    the default pattern is built from, symlinks included.
    """
    return os.path.realpath(path)

def jobs(patterns, max_version, max_rounds, analyses):
    """One job per (claim file, round count): the unit a single search covers.

    Each job is ``(cost, path, kind, rounds, label)``.
    """
    out = []
    for pattern in patterns:
        for path in sorted(glob.glob(pattern)):
            path = canonical(path)
            spec = load(path)
            data = spec["data"]
            claim = data["claim"]
            kind = claim["kind"]
            if analyses and claim["analysis"] not in analyses:
                continue
            width, label = width_of(spec), label_of(spec)
            if max_version and width and width > max_version:
                continue
            if kind == "trail":
                rounds = data.get("rounds") or data["target"].get("kwargs", {}).get("r") or 0
                if max_rounds and rounds > max_rounds:
                    continue
                out.append((cost(width, rounds), path, kind, rounds, label))
            else:
                for round_key in claim["values"]:
                    rounds = int(round_key)
                    if max_rounds and rounds > max_rounds:
                        continue
                    out.append((cost(width, rounds), path, kind, rounds, label))
    out.sort()
    return out

def done_keys(log, model_type):
    """Jobs already recorded for this backend.

    The key is ``(file, rounds, model_type)``, so switching from SAT to MILP
    (or vice versa) replays everything instead of silently reusing the other
    backend's results.
    """
    if not os.path.exists(log):
        return set()
    keys = set()
    with open(log, encoding="utf-8") as handle:
        for line in handle:
            line = line.strip()
            if not line:
                continue
            record = json.loads(line)
            keys.add((canonical(record["file"]), record["rounds"],
                       record.get("model_type", model_type)))
    return keys

def run_one(path, kind, rounds, model_type, config_solver, cache):
    """One job, as (checks, whether a search actually ran, the optimum found).

    The optimum is returned separately so the log can carry it as a number. It also appears inside
    each check's ``detail``, but that is prose for a human reading the run; ``make_table`` fills
    the table's OCP column from the field, rather than parsing a sentence that may be reworded.
    It is ``None`` for a trail claim, which pins a trail instead of searching for an optimum.

    Bound searches are shared through ``cache``: a target claimed by three papers is searched
    once and compared three times. Roughly half the recorded claims are such repeats.
    """
    spec = load(path)
    if kind == "trail":
        cipher = build_cipher(spec)
        results = verify(spec, cipher=cipher, model_type=model_type,
                         config_solver=config_solver)
        return [(r.name, r.passed, r.detail) for r in results], True, None
    key = search_key(spec, rounds)
    searched = key not in cache
    if searched:
        cache[key] = search_optimum(spec, rounds, model_type, config_solver)
    result = compare_bound(spec, rounds, cache[key])
    return [(result.name, result.passed, result.detail)], searched, cache[key]




def main():
    config_solver = {}
    if SOLVER:
        config_solver["solver"] = SOLVER
    if TIME_LIMIT:
        # Gurobi reads TimeLimit; SCIP and the SAT backends read time_limit.
        config_solver["TimeLimit"] = TIME_LIMIT
        config_solver["time_limit"] = TIME_LIMIT

    queue = jobs(PATTERNS, MAX_VERSION, MAX_ROUNDS, set(ANALYSES))
    already = done_keys(LOG, MODEL_TYPE)
    pending = [job for job in queue if (job[1], job[3], MODEL_TYPE) not in already]
    print(f"{len(queue)} jobs, {len(queue) - len(pending)} already in {LOG}, "
          f"{len(pending)} to run")
    if DRY_RUN:
        for _, path, kind, rounds, label in pending:
            print(f"  {label:<18} r{rounds:<3} {kind:<6} {path}")
        return 0

    failures, reused, cache = 0, 0, {}
    with open(LOG, "a", encoding="utf-8") as handle:
        for index, (_, path, kind, rounds, label) in enumerate(pending, 1):
            started = time.time()
            try:
                results, searched, optimum = run_one(path, kind, rounds, MODEL_TYPE,
                                                     config_solver, cache)
                error = None
            except Exception:                      # a broken claim must not stop the batch
                results, searched, optimum, error = [], True, None, traceback.format_exc()
            elapsed = round(time.time() - started, 2)
            reused += not searched
            passed = bool(results) and all(ok for _, ok, _ in results)
            failures += not passed
            record = {"file": path, "rounds": rounds, "target": label,
                      "model_type": MODEL_TYPE, "seconds": elapsed, "searched": searched,
                      "passed": passed, "optimum": optimum, "checks": results, "error": error}
            handle.write(json.dumps(record, ensure_ascii=False) + "\n")
            handle.flush()                         # so an interrupted run stays resumable
            mark = "ok  " if passed else "FAIL"
            if results:
                detail = "; ".join(d for _, _, d in results)
            else:
                detail = (error or "").strip().splitlines()[-1] if error else "no result"
            print(f"[{index}/{len(pending)}] {mark} {elapsed:>7.2f}s "
                  f"{'' if searched else '(cached) '}{os.path.basename(path)} r{rounds}: {detail}")
    print(f"\n{len(pending) - failures}/{len(pending)} passed; "
          f"{reused} of {len(pending)} answered from a shared search")
    return 1 if failures else 0


if __name__ == "__main__":
    main()
