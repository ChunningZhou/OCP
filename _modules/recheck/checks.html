Source code for recheck.checks

"""Replaying a published claim through OCP and reporting what the model says about it.

A trail claim is checked by four functions, cheapest first. What separates them is not the
question each asks -- each says that itself -- but two mechanical choices, what gets pinned and
what the solver is then asked for:

    feasible   pin the claim    ask for any solution
    weight     pin the claim    demand the claimed weight
    optimal    pin nothing      search for the optimum
    unique     pin nothing      search, forbid that optimum, search again

The pinned two stay cheap at high round counts because the trail fixes nearly every variable; the
searching two explore the whole space and are realistic only for few rounds.

A bound claim has no trail to pin, so it takes the other path: ``search_optimum`` for the round
count in question, then ``compare_bound`` against the claimed relation. That is **one full search
per round**, which is what makes replaying a bounds table expensive -- a ten-row table is ten
searches, not one.

``max_feasible_prefix`` is the diagnostic when a claim comes back infeasible: how far the
assignments can be pinned before the model refuses, which separates three causes that otherwise
look identical -- the paper is wrong, we transcribed it wrong, or our round/layer mapping is off.
"""
from collections import namedtuple

import attacks.attacks as attacks

from tools.model_constraints import gen_no_good_cut
from recheck.claim import (ROUND_KEY, assignments_to_solution, build_cipher,
                             pin_constraints, resolve_claim)

CheckResult = namedtuple("CheckResult", "name passed detail")

_ATTACK = {"differential": attacks.diff_attacks, "linear": attacks.linear_attacks}


def _search(spec, cipher, objective_target, constraints, model_type, config_solver):
    goal, _, weight_key = resolve_claim(spec)
    analysis = spec["data"]["claim"]["analysis"]
    trails = _ATTACK[analysis](
        cipher, goal=goal, constraints=constraints, objective_target=objective_target,
        show_mode=0, config_model={"model_type": model_type}, config_solver=config_solver or {},
    ) or []
    return trails, weight_key


def _claim_value(spec):
    return spec["data"]["claim"]["value"]


def _pins(spec, cipher, model_type, upto=None):
    """Constraints pinning the claim, optionally only its first ``upto`` assignments."""
    assignments = spec["data"].get("assignments", [])
    if upto is not None:
        assignments = assignments[:upto]
    return pin_constraints(assignments_to_solution(cipher, assignments), model_type)

def check_feasible(spec, cipher=None, model_type="milp", config_solver=None):
    """Does the model admit the published trail at all?"""
    cipher = cipher or build_cipher(spec)
    trails, _ = _search(spec, cipher, "EXISTENCE", _pins(spec, cipher, model_type),
                        model_type, config_solver)
    return CheckResult("feasible", bool(trails),
                       "admitted" if trails else "no solution with the claim pinned")




def _round_mismatch(trail, key, want_rounds):
    """Where the per-round series first disagrees, or None.

    Worth far more than the total: two rounds mis-transcribed in opposite directions leave the
    sum intact, so a matching total proves little, while a matching series localizes any error
    to one round.
    """
    got_rounds = trail.data.get(ROUND_KEY.get(key))
    if not got_rounds:
        return "the model reported no per-round series"
    if len(got_rounds) != len(want_rounds):
        return f"{len(got_rounds)} rounds reported, {len(want_rounds)} claimed"
    for i, (got, want) in enumerate(zip(got_rounds, want_rounds), start=1):
        if got != want:
            return f"round {i}: model {got} vs claimed {want}"
    return None

def check_weight(spec, cipher=None, model_type="milp", config_solver=None):
    """Does the published trail achieve the weight the paper states?

    When the claim also lists ``round_values`` -- the per-round figures papers usually print
    beside a trail -- those are compared one by one as well.
    """
    cipher = cipher or build_cipher(spec)
    want = _claim_value(spec)
    trails, key = _search(spec, cipher, f"EXACTLY {want}", _pins(spec, cipher, model_type),
                          model_type, config_solver)
    if not trails:
        return CheckResult("weight", False, f"not feasible at the claimed {want}")
    got = trails[0].data.get(key)
    if got != want:
        return CheckResult("weight", False, f"model {got} vs claimed {want}")
    want_rounds = spec["data"]["claim"].get("round_values")
    if not want_rounds:
        return CheckResult("weight", True, f"{got} as claimed (no per-round series to check)")
    bad = _round_mismatch(trails[0], key, want_rounds)
    return CheckResult("weight", bad is None,
                       f"{got} as claimed, per-round series matches" if bad is None
                       else f"total {got} matches but {bad}")

def check_optimal(spec, cipher=None, model_type="milp", config_solver=None):
    """Is the claimed value the best the model can do? Searches the whole space -- expensive."""
    cipher = cipher or build_cipher(spec)
    trails, key = _search(spec, cipher, "OPTIMAL", None, model_type, config_solver)
    if not trails:
        return CheckResult("optimal", False, "the unconstrained search found nothing")
    got, want = trails[0].data.get(key), _claim_value(spec)
    return CheckResult("optimal", got == want, f"optimum {got} vs claimed {want}")




def _no_good_cut(solution, model_type):
    """Forbid exactly this assignment, over the variables the solution pins to 0 or 1.

    The cut itself is ``tools.model_constraints.gen_no_good_cut``; this only selects which
    variables it covers. A value that is neither 0 nor 1 is left out rather than rounded in,
    which is what the copy this replaced did.
    """
    binary = [(k, int(round(float(v)))) for k, v in solution.items()]
    binary = [(k, b) for k, b in binary if b in (0, 1)]
    return gen_no_good_cut([k for k, _ in binary], [b for _, b in binary], model_type)

def check_unique(spec, cipher=None, model_type="milp", config_solver=None):
    """Is the optimum the only solution at that value? Expensive: two full searches."""
    cipher = cipher or build_cipher(spec)
    trails, key = _search(spec, cipher, "OPTIMAL", None, model_type, config_solver)
    if not trails:
        return CheckResult("unique", False, "the unconstrained search found nothing")
    best = trails[0].data.get(key)
    cut = _no_good_cut(trails[0].solution_trace, model_type)
    others, _ = _search(spec, cipher, f"EXACTLY {best}", cut, model_type, config_solver)
    return CheckResult("unique", not others,
                       f"optimum {best} is unique" if not others else f"another trail also reaches {best}")

def max_feasible_prefix(spec, cipher=None, model_type="milp", config_solver=None):
    """How many of the claim's assignments can be pinned before the model turns infeasible.

    Equal to the number of assignments when the whole claim is feasible; otherwise the index of
    the first one that cannot hold, which is where to look for the transcription error.
    """
    cipher = cipher or build_cipher(spec)
    total = len(spec["data"].get("assignments", []))
    for upto in range(1, total + 1):
        trails, _ = _search(spec, cipher, "EXISTENCE", _pins(spec, cipher, model_type, upto),
                            model_type, config_solver)
        if not trails:
            return upto - 1
    return total




RELATIONS = {"=": lambda got, want: got == want,
             ">=": lambda got, want: got >= want,
             "<=": lambda got, want: got <= want}

def search_key(spec, rounds):
    """What a bound search actually depends on, so identical searches can be shared.

    Several papers state the same optimum, and each is a separate claim file. The search behind
    them is one and the same: it is fixed by the primitive, the round count and the goal, and
    knows nothing of who claimed what. Batch runners key a cache on this.
    """
    target = spec["data"]["target"]
    claim = spec["data"]["claim"]
    kwargs = tuple(sorted((k, repr(v)) for k, v in target.get("kwargs", {}).items()))
    return (target["module"], target["constructor"], kwargs, rounds,
            claim["analysis"], claim.get("model"), claim["metric"])

def search_optimum(spec, rounds, model_type="milp", config_solver=None):
    """The model's optimal weight over ``rounds`` rounds, or None if nothing was found."""
    cipher = build_cipher(spec, rounds=rounds)
    trails, key = _search(spec, cipher, "OPTIMAL", None, model_type, config_solver)
    return trails[0].data.get(key) if trails else None

def compare_bound(spec, rounds, got):
    """Test an already-computed optimum against what the claim says about that round."""
    claim = spec["data"]["claim"]
    relation = claim.get("relation", "=")
    if relation not in RELATIONS:
        raise ValueError(f"Unknown relation {relation!r}. Expected one of {sorted(RELATIONS)}.")
    want = claim["values"][str(rounds)]
    ok = got is not None and RELATIONS[relation](got, want)
    return CheckResult(f"bound r{rounds}", ok, f"optimum {got} {relation} claimed {want}")

def check_bound(spec, model_type="milp", config_solver=None, rounds=None):
    """Search the optimum for each round the claim covers and test the stated relation.

    A bound claim carries no trail: it says what the best value over ``r`` rounds is (``"="``),
    or at least is (``">="``, how design documents usually phrase a security argument), or at
    most is (``"<="``, a trail was found but optimality was not proven). Nothing is pinned, so
    every round is a full search -- pass ``rounds`` to check only part of the table.
    """
    claim = spec["data"]["claim"]
    results = []
    for round_key in sorted(claim["values"], key=int):
        r = int(round_key)
        if rounds is not None and r not in rounds:
            continue
        got = search_optimum(spec, r, model_type, config_solver)
        results.append(compare_bound(spec, r, got))
    return results




CHECKS = {"feasible": check_feasible, "weight": check_weight,
          "optimal": check_optimal, "unique": check_unique}

def verify(spec, checks=("feasible", "weight"), cipher=None, model_type="milp", config_solver=None):
    """Run the named checks against one claim. Defaults to the two cheap ones."""
    cipher = cipher or build_cipher(spec)
    return [CHECKS[name](spec, cipher, model_type, config_solver) for name in checks]
