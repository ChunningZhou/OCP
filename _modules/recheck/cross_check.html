"""Checking published claims against each other, before any solver is asked to reproduce them."""
import glob
import sys
from collections import defaultdict
from pathlib import Path

# See reference/run.py: lets this run as `python reference/cross_check.py` as well as `-m`.
_ROOT = Path(__file__).resolve().parents[1]
if str(_ROOT) not in sys.path:
    sys.path.insert(0, str(_ROOT))

def collect(pattern):
    """(analysis, model, constructor, version, rounds) -> [(relation, value, filename)]."""
    table = defaultdict(list)
    for path in sorted(glob.glob(pattern)):
        spec = load(path)
        resolve_claim(spec)  # rejects an (analysis, model, metric) OCP has no goal for
        data = spec["data"]
        claim = data["claim"]
        if claim["kind"] != "bound":
            continue
        target = data["target"]
        version = version_label(target.get("kwargs", {}).get("version"))
        relation = claim.get("relation", "=")
        for rounds, value in claim["values"].items():
            key = (claim["analysis"], claim.get("model"), target["constructor"], version,
                   int(rounds))
            table[key].append((relation, float(value), path.split("/")[-1]))
    return table

def version_label(version):
    """A version kwarg as one hashable, sortable string.

    It arrives as an int for a permutation, as ``[block, key]`` for a block cipher, and as None
    for a primitive that takes no version at all, such as PRESENT and LBlock. Keeping those three
    shapes in a dict key makes it unhashable; keeping them in a sort key makes the comparison
    depend on which two entries happen to be adjacent.
    """
    if version is None:
        return ""
    if isinstance(version, (list, tuple)):
        return "-".join(str(part) for part in version)
    return str(version)




def _sortable(key):
    """A total order over keys, whose first four parts may be None or of mixed type."""
    return tuple(str(part) if part is not None else "" for part in key[:4]) + (key[4],)

def describe(key):
    """A key as one readable label: 'GIFT_PERMUTATION-64 r13 linear/characteristic'."""
    analysis, model, constructor, version, rounds = key
    name = constructor if not version else f"{constructor}-{version}"
    return f"{name} r{rounds} {analysis}/{model or 'sbox_count'}"

def conflicts(table):
    """Pairs of claims about the same target that cannot both hold.

    Two ``=`` claims must agree. An ``=`` claim of w and a ``<=`` claim of v require w <= v,
    since ``<=`` says a trail of weight v was found and w is the proved optimum, and an ``=`` of w
    with a ``>=`` of v requires w >= v. A ``>=`` above a ``<=`` is a contradiction on its own,
    with no optimum stated: no weight can be both at least 12 and at most 10.
    """
    found = []
    for key, claims in sorted(table.items(), key=lambda kv: _sortable(kv[0])):
        exact = [(v, f) for rel, v, f in claims if rel == "="]
        upper = [(v, f) for rel, v, f in claims if rel == "<="]
        lower = [(v, f) for rel, v, f in claims if rel == ">="]
        for i, (v1, f1) in enumerate(exact):
            for v2, f2 in exact[i + 1:]:
                if v1 != v2:
                    found.append((key, f"optima disagree: {v1} ({f1}) vs {v2} ({f2})"))
        for w, fw in exact:
            for v, fv in upper:
                if w > v:
                    found.append((key, f"optimum {w} ({fw}) exceeds the found weight {v} ({fv})"))
            for v, fv in lower:
                if w < v:
                    found.append((key, f"optimum {w} ({fw}) is below the lower bound {v} ({fv})"))
        for lo, flo in lower:
            for hi, fhi in upper:
                if lo > hi:
                    found.append((key, f"lower bound {lo} ({flo}) exceeds the found weight "
                                       f"{hi} ({fhi})"))
    return found

def pinned(table):
    """Targets where a ``>=`` and a ``<=`` meet, fixing the optimum without anyone searching.

    Two papers can settle a value between them: a design proof that no characteristic beats w,
    and somebody else's characteristic of weight w. The optimum is then w, and OCP reproducing it
    is a sharper test than either claim alone.
    """
    out = []
    for key, claims in table.items():
        lower = [(v, f) for rel, v, f in claims if rel == ">="]
        upper = [(v, f) for rel, v, f in claims if rel == "<="]
        for lo, flo in lower:
            for hi, fhi in upper:
                if lo == hi:
                    out.append((key, lo, flo, fhi))
    return sorted(out, key=lambda item: _sortable(item[0]))

def agreements(table):
    """Targets where independent sources state the same optimum -- the data's own evidence."""
    multi = [(key, sorted({f for rel, v, f in claims if rel == "="}))
             for key, claims in table.items()
             if len({f for rel, v, f in claims if rel == "="}) > 1]
    return sorted(multi, key=lambda item: _sortable(item[0]))




def report(pattern=None):
    table = collect(pattern or data_glob())
    bad = conflicts(table)
    same = agreements(table)
    print(f"{len(table)} distinct targets, {sum(map(len, table.values()))} claims")
    print(f"{len(same)} targets confirmed by more than one source")
    for key, sources in same:
        print(f"  {describe(key)}: {', '.join(sources)}")
    squeezed = pinned(table)
    if squeezed:
        print(f"{len(squeezed)} targets pinned between a lower and an upper bound")
        for key, value, flo, fhi in squeezed:
            print(f"  {describe(key)} = {value}: {flo} (>=) meets {fhi} (<=)")
    if bad:
        print(f"\n{len(bad)} CONFLICTS")
        for key, why in bad:
            print(f"  {describe(key)}: {why}")
    else:
        print("\nno conflicts")
    return bad


if __name__ == "__main__":
    sys.exit(1 if report() else 0)
