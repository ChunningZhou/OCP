def prefix_of(filename):
    for prefix in ORDER:
        if filename.startswith(prefix):
            return prefix
    return None


def cite(prefix):
    key = CITE[prefix]
    return f"\\cite{{{key}}}" if key else f"\\needbib{{{NEEDBIB[prefix]}}}"

def gather(pattern):
    """(analysis, model, version, relation, prefix) -> {rounds: value}, and the files behind each.

    ``sources`` is what lets the OCP column be filled: the replay log records a result per
    (claim file, round), while a row here may merge several files under one citation.
    """
    groups, sources = defaultdict(dict), defaultdict(set)
    skipped = []
    for path in sorted(glob.glob(pattern)):
        spec = load(path)
        data = spec["data"]
        claim = data["claim"]
        if claim["kind"] != "bound":
            continue
        name = path.split("/")[-1]
        prefix = prefix_of(name)
        if prefix is None or data.get("source") is None:
            skipped.append(name)
            continue
        key = (claim["analysis"], claim.get("model"),
               version_label(data["target"].get("kwargs", {}).get("version")),
               claim.get("relation", "="), prefix)
        groups[key].update({int(r): float(v) for r, v in claim["values"].items()})
        sources[key].add(name)
    return groups, sources, skipped

def replayed(log=RESULTS_LOG):
    """(claim filename, rounds) -> the optimum OCP found, from the replay log.

    Reads the ``optimum`` field rather than the check's prose. A record written before that field
    existed is skipped, so its row shows ``--`` until the claim is replayed again; the alternative,
    parsing "optimum 30.0 = claimed 30", pins the table to one wording of a sentence.
    """
    found = {}
    if not os.path.exists(log):
        return found
    with open(log, encoding="utf-8") as handle:
        for line in handle:
            line = line.strip()
            if not line:
                continue
            record = json.loads(line)
            if record.get("optimum") is None or record.get("rounds") is None:
                continue
            found[(os.path.basename(record["file"]), record["rounds"])] = record["optimum"]
    return found




def fmt(value):
    return str(int(value)) if float(value).is_integer() else f"{value:g}"

def ranges(rounds):
    """Collapse a sorted round list into '1--10' or '3, 5--7' form."""
    out, start, prev = [], rounds[0], rounds[0]
    for r in rounds[1:]:
        if r == prev + 1:
            prev = r
            continue
        out.append(f"{start}--{prev}" if prev > start else f"{start}")
        start = prev = r
    out.append(f"{start}--{prev}" if prev > start else f"{start}")
    return ", ".join(out)

def ocp_series(files, order, found):
    """The OCP column for one row: its own optima, ``--`` per round not replayed yet.

    ``None`` when no round of the row has been replayed, which the caller renders as a single
    ``---`` rather than a row of dashes as long as the claimed series.
    """
    got = [next((found[(f, r)] for f in sorted(files) if (f, r) in found), None) for r in order]
    if not any(value is not None for value in got):
        return None
    return ", ".join(fmt(value) if value is not None else "--" for value in got)




def rows(groups, sources, versions, found):
    for analysis, model in ANALYSES:
        block = []
        for version, display in versions:
            for key, values in sorted(groups.items(), key=lambda kv: kv[0][4]):
                if (key[0], key[1], key[2]) != (analysis, model, version):
                    continue
                relation, prefix = key[3], key[4]
                order = sorted(values)
                series = ", ".join(fmt(values[r]) for r in order)
                block.append(f"{display} & {ranges(order)} & {REL[relation]}{series} & "
                             f"{ocp_series(sources[key], order, found) or '---'} & "
                             f"{cite(prefix)} \\\\")
        if block:
            yield LABEL[(analysis, model)], block


def render(cipher):
    _, versions = CIPHERS[cipher]
    groups, sources, skipped = gather(data_glob(cipher))
    found = replayed()
    lines = ["\\begin{tabular}{llp{0.38\\textwidth}ll}",
             "\\toprule",
             "Version & Rounds & Claimed & OCP & Reference \\\\"]
    for label, block in rows(groups, sources, versions, found):
        lines.append("\\midrule")
        lines.append(f"\\multicolumn{{5}}{{l}}{{\\emph{{{label}}}}} \\\\")
        lines.append("\\cmidrule(lr){1-5}")
        lines.extend(block)
    lines += ["\\bottomrule", "\\end{tabular}"]
    return "\n".join(lines), skipped

def inject(cipher, path):
    """Replace the generated block inside ``path`` in place, leaving the prose around it alone.

    The table body lives in the section file rather than beside it, so the appendix is one file per
    cipher; the markers are what keeps it regenerable. A file without them is rejected rather than
    appended to, since guessing where the table belongs would silently produce a wrong document.
    """
    body, skipped = render(cipher)
    with open(path, encoding="utf-8") as handle:
        text = handle.read()
    if text.count(BEGIN) != 1 or text.count(END) != 1:
        raise SystemExit(f"{path}: expected exactly one {BEGIN!r} ... {END!r} block.")
    head, rest = text.split(BEGIN, 1)
    _, tail = rest.split(END, 1)
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(head + BEGIN + "\n" + body + "\n" + END + tail)
    return body.count("\\\\"), skipped




if __name__ == "__main__":
    which = sys.argv[1] if len(sys.argv) > 1 else "speck"
    if len(sys.argv) > 2:
        rows_written, skipped = inject(which, sys.argv[2])
        print(f"{sys.argv[2]}: {rows_written} rows")
    else:
        body, skipped = render(which)
        print(body)
    if skipped:
        print(f"% not published claims, excluded: {', '.join(skipped)}")
